  1. لَهُۥ مَا فِى ٱلسَّمَـٰوَٰتِ وَمَا فِى ٱلْأَرْضِ وَمَا بَيْنَهُمَا وَمَا تَحْتَ ٱلثَّرَىٰ ﴿٦﴾

    الشورى ٤

    لَهُۥ مَا فِى ٱلسَّمَـٰوَٰتِ وَمَا فِى ٱلْأَرْضِ ۖ وَهُوَ ٱلْعَلِىُّ ٱلْعَظِيمُ ﴿٤﴾

    «وَهُوَ ٱلْعَلِىُّ ٱلْعَظِيمُ» بدل «وَمَا بَيْنَهُمَا وَمَا تَحْتَ ٱلثَّرَىٰ»

  2. ٱللَّهُ لَآ إِلَـٰهَ إِلَّا هُوَ ۖ لَهُ ٱلْأَسْمَآءُ ٱلْحُسْنَىٰ ﴿٨﴾

    آل عمران ٢

    ٱللَّهُ لَآ إِلَـٰهَ إِلَّا هُوَ ٱلْحَىُّ ٱلْقَيُّومُ ﴿٢﴾

    «ٱلْحَىُّ ٱلْقَيُّومُ» بدل «لَهُ ٱلْأَسْمَآءُ ٱلْحُسْنَىٰ»

    وَهُوَ ٱللَّهُ لَآ إِلَـٰهَ إِلَّا هُوَ ۖ لَهُ ٱلْحَمْدُ فِى ٱلْأُولَىٰ وَٱلْـَٔاخِرَةِ ۖ وَلَهُ ٱلْحُكْمُ وَإِلَيْهِ تُرْجَعُونَ ﴿٧٠﴾

    قَالُوا۟ يَـٰمُوسَىٰٓ إِمَّآ أَن تُلْقِىَ وَإِمَّآ أَن نَّكُونَ نَحْنُ ٱلْمُلْقِينَ ﴿١١٥﴾

    الأعراف ١٢٤

    لَأُقَطِّعَنَّ أَيْدِيَكُمْ وَأَرْجُلَكُم مِّنْ خِلَـٰفٍ ثُمَّ لَأُصَلِّبَنَّكُمْ أَجْمَعِينَ ﴿١٢٤﴾

    «لَأُقَطِّعَنَّ» بدل «قَالَ ءَامَنتُمْ لَهُۥ … ٱلسِّحْرَ فَلَأُقَطِّعَنَّ»؛ «ثُمَّ لَأُصَلِّبَنَّكُمْ أَجْمَعِينَ» بدل «وَلَأُصَلِّبَنَّكُمْ فِى جُذُوعِ … عَذَابًا وَأَبْقَىٰ»

  9. إِنَّهُۥ مَن يَأْتِ رَبَّهُۥ مُجْرِمًا فَإِنَّ لَهُۥ جَهَنَّمَ لَا يَمُوتُ فِيهَا وَلَا يَحْيَىٰ ﴿٧٤﴾

    الأعلى ١٣

    ثُمَّ لَا يَمُوتُ فِيهَا وَلَا يَحْيَىٰ ﴿١٣﴾

    «ثُمَّ» بدل «إِنَّهُۥ مَن يَأْتِ رَبَّهُۥ مُجْرِمًا فَإِنَّ لَهُۥ جَهَنَّمَ»

  10. جَنَّـٰتُ عَدْنٍ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَـٰرُ خَـٰلِدِينَ فِيهَا ۚ وَذَٰلِكَ جَزَآءُ مَن تَزَكَّىٰ ﴿٧٦﴾

    المائدة ٨٥

    فَأَثَـٰبَهُمُ ٱللَّهُ بِمَا قَالُوا۟ جَنَّـٰتٍ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَـٰرُ خَـٰلِدِينَ فِيهَا ۚ وَذَٰلِكَ جَزَآءُ ٱلْمُحْسِنِينَ ﴿٨٥﴾

    «فَأَثَـٰبَهُمُ ٱللَّهُ بِمَا قَالُوا۟ جَنَّـٰتٍ» بدل «جَنَّـٰتُ عَدْنٍ»؛ «ٱلْمُحْسِنِينَ» بدل «مَن تَزَكَّىٰ»

    البينة ٨

    جَزَآؤُهُمْ عِندَ رَبِّهِمْ جَنَّـٰتُ عَدْنٍ تَجْرِى مِن تَحْتِهَا ٱلْأَنْهَـٰرُ خَـٰلِدِينَ فِيهَآ أَبَدًا ۖ رَّضِىَ ٱللَّهُ عَنْهُمْ وَرَضُوا۟ عَنْهُ ۚ ذَٰلِكَ لِمَنْ خَشِىَ رَبَّهُۥ ﴿٨﴾

    آخرها يشبه آية هذه السورة: «أَبَدًا رَّضِىَ ٱللَّهُ … خَشِىَ رَبَّهُۥ» بدل «وَذَٰلِكَ جَزَآءُ مَن تَزَكَّىٰ»

  11. وَلَقَدْ أَوْحَيْنَآ إِلَىٰ مُوسَىٰٓ أَنْ أَسْرِ بِعِبَادِى فَٱضْرِبْ لَهُمْ طَرِيقًا فِى ٱلْبَحْرِ يَبَسًا لَّا تَخَـٰفُ دَرَكًا وَلَا تَخْشَىٰ ﴿٧٧﴾

    الشعراء ٥٢

    وَأَوْحَيْنَآ إِلَىٰ مُوسَىٰٓ أَنْ أَسْرِ بِعِبَادِىٓ إِنَّكُم مُّتَّبَعُونَ ﴿٥٢﴾

    «وَأَوْحَيْنَآ» بدل «وَلَقَدْ أَوْحَيْنَآ»؛ «إِنَّكُم مُّتَّبَعُونَ» بدل «فَٱضْرِبْ لَهُمْ طَرِيقًا … وَلَا تَخْشَىٰ»

  12. إِنَّمَآ إِلَـٰهُكُمُ ٱللَّهُ ٱلَّذِى لَآ إِلَـٰهَ إِلَّا هُوَ ۚ وَسِعَ كُلَّ شَىْءٍ عِلْمًا ﴿٩٨﴾

    آل عمران ٢

    ٱللَّهُ لَآ إِلَـٰهَ إِلَّا هُوَ ٱلْحَىُّ ٱلْقَيُّومُ ﴿٢﴾

    تشبه آخرَ آية هذه السورة: حذف «ٱلَّذِى»؛ «ٱلْحَىُّ ٱلْقَيُّومُ» بدل «وَسِعَ كُلَّ شَىْءٍ عِلْمًا»

  13. يَوْمَ يُنفَخُ فِى ٱلصُّورِ ۚ وَنَحْشُرُ ٱلْمُجْرِمِينَ يَوْمَئِذٍ زُرْقًا ﴿١٠٢﴾

    النبإ ١٨

    يَوْمَ يُنفَخُ فِى ٱلصُّورِ فَتَأْتُونَ أَفْوَاجًا ﴿١٨﴾

    «فَتَأْتُونَ أَفْوَاجًا» بدل «وَنَحْشُرُ ٱلْمُجْرِمِينَ يَوْمَئِذٍ زُرْقًا»

  14. يَعْلَمُ مَا بَيْنَ أَيْدِيهِمْ وَمَا خَلْفَهُمْ وَلَا يُحِيطُونَ بِهِۦ عِلْمًا ﴿١١٠﴾

    البقرة ٢٥٥

    ٱللَّهُ لَآ إِلَـٰهَ إِلَّا هُوَ ٱلْحَىُّ ٱلْقَيُّومُ ۚ لَا تَأْخُذُهُۥ سِنَةٌ وَلَا نَوْمٌ ۚ لَّهُۥ مَا فِى ٱلسَّمَـٰوَٰتِ وَمَا فِى ٱلْأَرْضِ ۗ مَن ذَا ٱلَّذِى يَشْفَعُ عِندَهُۥٓ إِلَّا بِإِذْنِهِۦ ۚ يَعْلَمُ مَا بَيْنَ أَيْدِيهِمْ وَمَا خَلْفَهُمْ ۖ وَلَا يُحِيطُونَ بِشَىْءٍ مِّنْ عِلْمِهِۦٓ إِلَّا بِمَا شَآءَ ۚ وَسِعَ كُرْسِيُّهُ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضَ ۖ وَلَا يَـُٔودُهُۥ حِفْظُهُمَا ۚ وَهُوَ ٱلْعَلِىُّ ٱلْعَظِيمُ ﴿٢٥٥﴾

    زيادة «ٱللَّهُ لَآ إِلَـٰهَ … إِلَّا بِإِذْنِهِۦ» في أولها؛ «بِشَىْءٍ مِّنْ عِلْمِهِۦٓ … ٱلْعَلِىُّ ٱلْعَظِيمُ» بدل «بِهِۦ عِلْمًا»

    الأنبياء ٢٨

    يَعْلَمُ مَا بَيْنَ أَيْدِيهِمْ وَمَا خَلْفَهُمْ وَلَا يَشْفَعُونَ إِلَّا لِمَنِ ٱرْتَضَىٰ وَهُم مِّنْ خَشْيَتِهِۦ مُشْفِقُونَ ﴿٢٨﴾

    «يَشْفَعُونَ إِلَّا لِمَنِ ٱرْتَضَىٰ وَهُم مِّنْ خَشْيَتِهِۦ مُشْفِقُونَ» بدل «يُحِيطُونَ بِهِۦ عِلْمًا»

  15. وَإِذْ قُلْنَا لِلْمَلَـٰٓئِكَةِ ٱسْجُدُوا۟ لِـَٔادَمَ فَسَجَدُوٓا۟ إِلَّآ إِبْلِيسَ أَبَىٰ ﴿١١٦﴾

    البقرة ٣٤

    وَإِذْ قُلْنَا لِلْمَلَـٰٓئِكَةِ ٱسْجُدُوا۟ لِـَٔادَمَ فَسَجَدُوٓا۟ إِلَّآ إِبْلِيسَ أَبَىٰ وَٱسْتَكْبَرَ وَكَانَ مِنَ ٱلْكَـٰفِرِينَ ﴿٣٤﴾

    زيادة «وَٱسْتَكْبَرَ وَكَانَ مِنَ ٱلْكَـٰفِرِينَ» في آخرها

    الأعراف ١١

    وَلَقَدْ خَلَقْنَـٰكُمْ ثُمَّ صَوَّرْنَـٰكُمْ ثُمَّ قُلْنَا لِلْمَلَـٰٓئِكَةِ ٱسْجُدُوا۟ لِـَٔادَمَ فَسَجَدُوٓا۟ إِلَّآ إِبْلِيسَ لَمْ يَكُن مِّنَ ٱلسَّـٰجِدِينَ ﴿١١﴾

    «وَلَقَدْ خَلَقْنَـٰكُمْ ثُمَّ صَوَّرْنَـٰكُمْ ثُمَّ» بدل «وَإِذْ»؛ «لَمْ يَكُن مِّنَ ٱلسَّـٰجِدِينَ» بدل «أَبَىٰ»

    الإسراء ٦١

    وَإِذْ قُلْنَا لِلْمَلَـٰٓئِكَةِ ٱسْجُدُوا۟ لِـَٔادَمَ فَسَجَدُوٓا۟ إِلَّآ إِبْلِيسَ قَالَ ءَأَسْجُدُ لِمَنْ خَلَقْتَ طِينًا ﴿٦١﴾

    «قَالَ ءَأَسْجُدُ لِمَنْ خَلَقْتَ طِينًا» بدل «أَبَىٰ»

    الكهف ٥٠

    وَإِذْ قُلْنَا لِلْمَلَـٰٓئِكَةِ ٱسْجُدُوا۟ لِـَٔادَمَ فَسَجَدُوٓا۟ إِلَّآ إِبْلِيسَ كَانَ مِنَ ٱلْجِنِّ فَفَسَقَ عَنْ أَمْرِ رَبِّهِۦٓ ۗ أَفَتَتَّخِذُونَهُۥ وَذُرِّيَّتَهُۥٓ أَوْلِيَآءَ مِن دُونِى وَهُمْ لَكُمْ عَدُوٌّۢ ۚ بِئْسَ لِلظَّـٰلِمِينَ بَدَلًا ﴿٥٠﴾

    «كَانَ مِنَ ٱلْجِنِّ … لِلظَّـٰلِمِينَ بَدَلًا» بدل «أَبَىٰ»

  16. أَفَلَمْ يَهْدِ لَهُمْ كَمْ أَهْلَكْنَا قَبْلَهُم مِّنَ ٱلْقُرُونِ يَمْشُونَ فِى مَسَـٰكِنِهِمْ ۗ إِنَّ فِى ذَٰلِكَ لَـَٔايَـٰتٍ لِّأُو۟لِى ٱلنُّهَىٰ ﴿١٢٨﴾

    الحجر ٧٥

    إِنَّ فِى ذَٰلِكَ لَـَٔايَـٰتٍ لِّلْمُتَوَسِّمِينَ ﴿٧٥﴾

    تشبه آخرَ آية هذه السورة: «لِّلْمُتَوَسِّمِينَ» بدل «لِّأُو۟لِى ٱلنُّهَىٰ»

    السجدة ٢٦

    أَوَلَمْ يَهْدِ لَهُمْ كَمْ أَهْلَكْنَا مِن قَبْلِهِم مِّنَ ٱلْقُرُونِ يَمْشُونَ فِى مَسَـٰكِنِهِمْ ۚ إِنَّ فِى ذَٰلِكَ لَـَٔايَـٰتٍ ۖ أَفَلَا يَسْمَعُونَ ﴿٢٦﴾

    «أَوَلَمْ» بدل «أَفَلَمْ»؛ «مِن قَبْلِهِم» بدل «قَبْلَهُم»؛ «أَفَلَا يَسْمَعُونَ» بدل «لِّأُو۟لِى ٱلنُّهَىٰ»

  17. وَلَوْلَا كَلِمَةٌ سَبَقَتْ مِن رَّبِّكَ لَكَانَ لِزَامًا وَأَجَلٌ مُّسَمًّى ﴿١٢٩﴾

    الشورى ١٤

    وَمَا تَفَرَّقُوٓا۟ إِلَّا مِنۢ بَعْدِ مَا جَآءَهُمُ ٱلْعِلْمُ بَغْيًۢا بَيْنَهُمْ ۚ وَلَوْلَا كَلِمَةٌ سَبَقَتْ مِن رَّبِّكَ إِلَىٰٓ أَجَلٍ مُّسَمًّى لَّقُضِىَ بَيْنَهُمْ ۚ وَإِنَّ ٱلَّذِينَ أُورِثُوا۟ ٱلْكِتَـٰبَ مِنۢ بَعْدِهِمْ لَفِى شَكٍّ مِّنْهُ مُرِيبٍ ﴿١٤﴾

    في وسطها ما يشبه آية هذه السورة: «إِلَىٰٓ أَجَلٍ» بدل «لَكَانَ لِزَامًا وَأَجَلٌ»

  18. فَٱصْبِرْ عَلَىٰ مَا يَقُولُونَ وَسَبِّحْ بِحَمْدِ رَبِّكَ قَبْلَ طُلُوعِ ٱلشَّمْسِ وَقَبْلَ غُرُوبِهَا ۖ وَمِنْ ءَانَآئِ ٱلَّيْلِ فَسَبِّحْ وَأَطْرَافَ ٱلنَّهَارِ لَعَلَّكَ تَرْضَىٰ ﴿١٣٠﴾

    ق ٣٩

    فَٱصْبِرْ عَلَىٰ مَا يَقُولُونَ وَسَبِّحْ بِحَمْدِ رَبِّكَ قَبْلَ طُلُوعِ ٱلشَّمْسِ وَقَبْلَ ٱلْغُرُوبِ ﴿٣٩﴾

    «ٱلْغُرُوبِ» بدل «غُرُوبِهَا وَمِنْ ءَانَآئِ … لَعَلَّكَ تَرْضَىٰ»

    «أَوَلَمْ» en lugar de «أَفَلَمْ»; «مِن قَبْلِهِم» en lugar de «قَبْلَهُم»; «أَفَلَا يَسْمَعُونَ» en lugar de «لِّأُو۟لِى ٱلنُّهَىٰ»

  17. وَلَوْلَا كَلِمَةٌ سَبَقَتْ مِن رَّبِّكَ لَكَانَ لِزَامًا وَأَجَلٌ مُّسَمًّى ﴿١٢٩﴾

    الشورى 14

    وَمَا تَفَرَّقُوٓا۟ إِلَّا مِنۢ بَعْدِ مَا جَآءَهُمُ ٱلْعِلْمُ بَغْيًۢا بَيْنَهُمْ ۚ وَلَوْلَا كَلِمَةٌ سَبَقَتْ مِن رَّبِّكَ إِلَىٰٓ أَجَلٍ مُّسَمًّى لَّقُضِىَ بَيْنَهُمْ ۚ وَإِنَّ ٱلَّذِينَ أُورِثُوا۟ ٱلْكِتَـٰبَ مِنۢ بَعْدِهِمْ لَفِى شَكٍّ مِّنْهُ مُرِيبٍ ﴿١٤﴾

    Su parte central se parece a la aleya de esta sura: «إِلَىٰٓ أَجَلٍ» en lugar de «لَكَانَ لِزَامًا وَأَجَلٌ»

  18. فَٱصْبِرْ عَلَىٰ مَا يَقُولُونَ وَسَبِّحْ بِحَمْدِ رَبِّكَ قَبْلَ طُلُوعِ ٱلشَّمْسِ وَقَبْلَ غُرُوبِهَا ۖ وَمِنْ ءَانَآئِ ٱلَّيْلِ فَسَبِّحْ وَأَطْرَافَ ٱلنَّهَارِ لَعَلَّكَ تَرْضَىٰ ﴿١٣٠﴾

    ق 39

    فَٱصْبِرْ عَلَىٰ مَا يَقُولُونَ وَسَبِّحْ بِحَمْدِ رَبِّكَ قَبْلَ طُلُوعِ ٱلشَّمْسِ وَقَبْلَ ٱلْغُرُوبِ ﴿٣٩﴾
