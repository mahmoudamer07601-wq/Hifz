  1. ١ آيتان تشتركان في «ٱتَّخَذُوا۟ مِن دُونِهِۦٓ أَوْلِيَآءَ»

    وَٱلَّذِينَ ٱتَّخَذُوا۟ مِن دُونِهِۦٓ أَوْلِيَآءَ ٱللَّهُ حَفِيظٌ عَلَيْهِمْ وَمَآ أَنتَ عَلَيْهِم بِوَكِيلٍ ﴿٦﴾

    أَمِ ٱتَّخَذُوا۟ مِن دُونِهِۦٓ أَوْلِيَآءَ ۖ فَٱللَّهُ هُوَ ٱلْوَلِىُّ وَهُوَ يُحْىِ ٱلْمَوْتَىٰ وَهُوَ عَلَىٰ كُلِّ شَىْءٍ قَدِيرٌ ﴿٩﴾

    قبلها «أَمِ» بدل «وَٱلَّذِينَ»؛ بعدها «فَٱللَّهُ» بدل «ٱللَّهُ»

  2. ٢ آيتان تشتركان في «وَمَن يُضْلِلِ ٱللَّهُ فَمَا لَهُۥ مِن»

    وَمَن يُضْلِلِ ٱللَّهُ فَمَا لَهُۥ مِن وَلِىٍّ مِّنۢ بَعْدِهِۦ ۗ وَتَرَى … ﴿٤٤﴾

    وَمَا كَانَ لَهُم مِّنْ أَوْلِيَآءَ يَنصُرُونَهُم مِّن دُونِ ٱللَّهِ ۗ وَمَن يُضْلِلِ ٱللَّهُ فَمَا لَهُۥ مِن سَبِيلٍ ﴿٤٦﴾

    قبلها «ٱللَّهِ»؛ بعدها «سَبِيلٍ» بدل «وَلِىٍّ»

آيات تشبهها في سور أخرى

  1. لَهُۥ مَا فِى ٱلسَّمَـٰوَٰتِ وَمَا فِى ٱلْأَرْضِ ۖ وَهُوَ ٱلْعَلِىُّ ٱلْعَظِيمُ ﴿٤﴾

    البقرة ٢٥٥

    ٱللَّهُ لَآ إِلَـٰهَ إِلَّا هُوَ ٱلْحَىُّ ٱلْقَيُّومُ ۚ لَا تَأْخُذُهُۥ سِنَةٌ وَلَا نَوْمٌ ۚ لَّهُۥ مَا فِى ٱلسَّمَـٰوَٰتِ وَمَا فِى ٱلْأَرْضِ ۗ مَن ذَا ٱلَّذِى يَشْفَعُ عِندَهُۥٓ إِلَّا بِإِذْنِهِۦ ۚ يَعْلَمُ مَا بَيْنَ أَيْدِيهِمْ وَمَا خَلْفَهُمْ ۖ وَلَا يُحِيطُونَ بِشَىْءٍ مِّنْ عِلْمِهِۦٓ إِلَّا بِمَا شَآءَ ۚ وَسِعَ كُرْسِيُّهُ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضَ ۖ وَلَا يَـُٔودُهُۥ حِفْظُهُمَا ۚ وَهُوَ ٱلْعَلِىُّ ٱلْعَظِيمُ ﴿٢٥٥﴾

    آخرها يشبه آية هذه السورة: زيادة «مَن ذَا ٱلَّذِى … يَـُٔودُهُۥ حِفْظُهُمَا»

    النساء ١٧١

    يَـٰٓأَهْلَ ٱلْكِتَـٰبِ لَا تَغْلُوا۟ فِى دِينِكُمْ وَلَا تَقُولُوا۟ عَلَى ٱللَّهِ إِلَّا ٱلْحَقَّ ۚ إِنَّمَا ٱلْمَسِيحُ عِيسَى ٱبْنُ مَرْيَمَ رَسُولُ ٱللَّهِ وَكَلِمَتُهُۥٓ أَلْقَىٰهَآ إِلَىٰ مَرْيَمَ وَرُوحٌ مِّنْهُ ۖ فَـَٔامِنُوا۟ بِٱللَّهِ وَرُسُلِهِۦ ۖ وَلَا تَقُولُوا۟ ثَلَـٰثَةٌ ۚ ٱنتَهُوا۟ خَيْرًا لَّكُمْ ۚ إِنَّمَا ٱللَّهُ إِلَـٰهٌ وَٰحِدٌ ۖ سُبْحَـٰنَهُۥٓ أَن يَكُونَ لَهُۥ وَلَدٌ ۘ لَّهُۥ مَا فِى ٱلسَّمَـٰوَٰتِ وَمَا فِى ٱلْأَرْضِ ۗ وَكَفَىٰ بِٱللَّهِ وَكِيلًا ﴿١٧١﴾

    آخرها يشبه آية هذه السورة: «وَكَفَىٰ بِٱللَّهِ وَكِيلًا» بدل «وَهُوَ ٱلْعَلِىُّ ٱلْعَظِيمُ»

    يونس ٦٨

    قَالُوا۟ ٱتَّخَذَ ٱللَّهُ وَلَدًا ۗ سُبْحَـٰنَهُۥ ۖ هُوَ ٱلْغَنِىُّ ۖ لَهُۥ مَا فِى ٱلسَّمَـٰوَٰتِ وَمَا فِى ٱلْأَرْضِ ۚ إِنْ عِندَكُم مِّن سُلْطَـٰنٍۭ بِهَـٰذَآ ۚ أَتَقُولُونَ عَلَى ٱللَّهِ مَا لَا تَعْلَمُونَ ﴿٦٨﴾

    زيادة «قَالُوا۟ ٱتَّخَذَ ٱللَّهُ وَلَدًا سُبْحَـٰنَهُۥ هُوَ ٱلْغَنِىُّ» في أولها؛ «إِنْ عِندَكُم مِّن … لَا تَعْلَمُونَ» بدل «وَهُوَ ٱلْعَلِىُّ ٱلْعَظِيمُ»

    ابراهيم ٢

    ٱللَّهِ ٱلَّذِى لَهُۥ مَا فِى ٱلسَّمَـٰوَٰتِ وَمَا فِى ٱلْأَرْضِ ۗ وَوَيْلٌ لِّلْكَـٰفِرِينَ مِنْ عَذَابٍ شَدِيدٍ ﴿٢﴾

    آخرها يشبه آية هذه السورة: «وَوَيْلٌ لِّلْكَـٰفِرِينَ مِنْ عَذَابٍ شَدِيدٍ» بدل «وَهُوَ ٱلْعَلِىُّ ٱلْعَظِيمُ»

    طه ٦

    لَهُۥ مَا فِى ٱلسَّمَـٰوَٰتِ وَمَا فِى ٱلْأَرْضِ وَمَا بَيْنَهُمَا وَمَا تَحْتَ ٱلثَّرَىٰ ﴿٦﴾

    «وَمَا بَيْنَهُمَا وَمَا تَحْتَ ٱلثَّرَىٰ» بدل «وَهُوَ ٱلْعَلِىُّ ٱلْعَظِيمُ»

    الحج ٦٤

    لَّهُۥ مَا فِى ٱلسَّمَـٰوَٰتِ وَمَا فِى ٱلْأَرْضِ ۗ وَإِنَّ ٱللَّهَ لَهُوَ ٱلْغَنِىُّ ٱلْحَمِيدُ ﴿٦٤﴾

    «وَإِنَّ ٱللَّهَ لَهُوَ ٱلْغَنِىُّ ٱلْحَمِيدُ» بدل «وَهُوَ ٱلْعَلِىُّ ٱلْعَظِيمُ»

    سبإ ١

    ٱلْحَمْدُ لِلَّهِ ٱلَّذِى لَهُۥ مَا فِى ٱلسَّمَـٰوَٰتِ وَمَا فِى ٱلْأَرْضِ وَلَهُ ٱلْحَمْدُ فِى ٱلْـَٔاخِرَةِ ۚ وَهُوَ ٱلْحَكِيمُ ٱلْخَبِيرُ ﴿١﴾

    آخرها يشبه آية هذه السورة: «وَلَهُ ٱلْحَمْدُ فِى ٱلْـَٔاخِرَةِ وَهُوَ ٱلْحَكِيمُ ٱلْخَبِيرُ» بدل «وَهُوَ ٱلْعَلِىُّ ٱلْعَظِيمُ»

    الحشر ١؛ الصف ١

    سَبَّحَ لِلَّهِ مَا فِى ٱلسَّمَـٰوَٰتِ وَمَا فِى ٱلْأَرْضِ ۖ وَهُوَ ٱلْعَزِيزُ ٱلْحَكِيمُ ﴿١﴾

    «سَبَّحَ لِلَّهِ» بدل «لَهُۥ»؛ «ٱلْعَزِيزُ ٱلْحَكِيمُ» بدل «ٱلْعَلِىُّ ٱلْعَظِيمُ»

  2. وَمَا تَفَرَّقُوٓا۟ إِلَّا مِنۢ بَعْدِ مَا جَآءَهُمُ ٱلْعِلْمُ بَغْيًۢا بَيْنَهُمْ ۚ وَلَوْلَا كَلِمَةٌ سَبَقَتْ مِن رَّبِّكَ إِلَىٰٓ أَجَلٍ مُّسَمًّى لَّقُضِىَ بَيْنَهُمْ ۚ وَإِنَّ ٱلَّذِينَ أُورِثُوا۟ ٱلْكِتَـٰبَ مِنۢ بَعْدِهِمْ لَفِى شَكٍّ مِّنْهُ مُرِيبٍ ﴿١٤﴾

    وَلَوْلَا كَلِمَةٌ سَبَقَتْ مِن رَّبِّكَ لَكَانَ لِزَامًا وَأَجَلٌ مُّسَمًّى ﴿١٢٩﴾

    تشبه جزءاً من آية هذه السورة: «لَكَانَ لِزَامًا وَأَجَلٌ» بدل «إِلَىٰٓ أَجَلٍ»

  3. تَرَى ٱلظَّـٰلِمِينَ مُشْفِقِينَ مِمَّا كَسَبُوا۟ وَهُوَ وَاقِعٌۢ بِهِمْ ۗ وَٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّـٰلِحَـٰتِ فِى رَوْضَاتِ ٱلْجَنَّاتِ ۖ لَهُم مَّا يَشَآءُونَ عِندَ رَبِّهِمْ ۚ ذَٰلِكَ هُوَ ٱلْفَضْلُ ٱلْكَبِيرُ ﴿٢٢﴾

    العنكبوت ٩

    وَٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّـٰلِحَـٰتِ لَنُدْخِلَنَّهُمْ فِى ٱلصَّـٰلِحِينَ ﴿٩﴾

    حذف «تَرَى ٱلظَّـٰلِمِينَ مُشْفِقِينَ مِمَّا كَسَبُوا۟ وَهُوَ وَاقِعٌۢ بِهِمْ» من أولها؛ «لَنُدْخِلَنَّهُمْ فِى ٱلصَّـٰلِحِينَ» بدل «فِى رَوْضَاتِ ٱلْجَنَّاتِ … ٱلْفَضْلُ ٱلْكَبِيرُ»

    الزمر ٣٤

    لَهُم مَّا يَشَآءُونَ عِندَ رَبِّهِمْ ۚ ذَٰلِكَ جَزَآءُ ٱلْمُحْسِنِينَ ﴿٣٤﴾

    تشبه آخرَ آية هذه السورة: «جَزَآءُ ٱلْمُحْسِنِينَ» بدل «هُوَ ٱلْفَضْلُ ٱلْكَبِيرُ»

  4. وَمَآ أَنتُم بِمُعْجِزِينَ فِى ٱلْأَرْضِ ۖ وَمَا لَكُم مِّن دُونِ ٱللَّهِ مِن وَلِىٍّ وَلَا نَصِيرٍ ﴿٣١﴾

    البقرة ١٠٧

    أَلَمْ تَعْلَمْ أَنَّ ٱللَّهَ لَهُۥ مُلْكُ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ ۗ وَمَا لَكُم مِّن دُونِ ٱللَّهِ مِن وَلِىٍّ وَلَا نَصِيرٍ ﴿١٠٧﴾

    «أَلَمْ تَعْلَمْ أَنَّ ٱللَّهَ لَهُۥ مُلْكُ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ» بدل «وَمَآ أَنتُم بِمُعْجِزِينَ فِى ٱلْأَرْضِ»

    العنكبوت ٢٢

    وَمَآ أَنتُم بِمُعْجِزِينَ فِى ٱلْأَرْضِ وَلَا فِى ٱلسَّمَآءِ ۖ وَمَا لَكُم مِّن دُونِ ٱللَّهِ مِن وَلِىٍّ وَلَا نَصِيرٍ ﴿٢٢﴾

    زيادة «وَلَا فِى ٱلسَّمَآءِ»

  5. وَمِنْ ءَايَـٰتِهِ ٱلْجَوَارِ فِى ٱلْبَحْرِ كَٱلْأَعْلَـٰمِ ﴿٣٢﴾

    الرحمن ٢٤

    وَلَهُ ٱلْجَوَارِ ٱلْمُنشَـَٔاتُ فِى ٱلْبَحْرِ كَٱلْأَعْلَـٰمِ ﴿٢٤﴾

    «وَلَهُ ٱلْجَوَارِ ٱلْمُنشَـَٔاتُ» بدل «وَمِنْ ءَايَـٰتِهِ ٱلْجَوَارِ»

  6. إِن يَشَأْ يُسْكِنِ ٱلرِّيحَ فَيَظْلَلْنَ رَوَاكِدَ عَلَىٰ ظَهْرِهِۦٓ ۚ إِنَّ فِى ذَٰلِكَ لَـَٔايَـٰتٍ لِّكُلِّ صَبَّارٍ شَكُورٍ ﴿٣٣﴾

    الحجر ٧٥

    إِنَّ فِى ذَٰلِكَ لَـَٔايَـٰتٍ لِّلْمُتَوَسِّمِينَ ﴿٧٥﴾

    تشبه آخرَ آية هذه السورة: «لِّلْمُتَوَسِّمِينَ» بدل «لِّكُلِّ صَبَّارٍ شَكُورٍ»

  7. وَكَذَٰلِكَ أَوْحَيْنَآ إِلَيْكَ رُوحًا مِّنْ أَمْرِنَا ۚ مَا كُنتَ تَدْرِى مَا ٱلْكِتَـٰبُ وَلَا ٱلْإِيمَـٰنُ وَلَـٰكِن جَعَلْنَـٰهُ نُورًا نَّهْدِى بِهِۦ مَن نَّشَآءُ مِنْ عِبَادِنَا ۚ وَإِنَّكَ لَتَهْدِىٓ إِلَىٰ صِرَٰطٍ مُّسْتَقِيمٍ ﴿٥٢﴾

    المؤمنون ٧٣

    وَإِنَّكَ لَتَدْعُوهُمْ إِلَىٰ صِرَٰطٍ مُّسْتَقِيمٍ ﴿٧٣﴾

    تشبه آخرَ آية هذه السورة: «لَتَدْعُوهُمْ» بدل «لَتَهْدِىٓ»
