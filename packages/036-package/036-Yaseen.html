  1. تَنزِيلَ ٱلْعَزِيزِ ٱلرَّحِيمِ ﴿٥﴾

    فصلت ٢

  2. وَسَوَآءٌ عَلَيْهِمْ ءَأَنذَرْتَهُمْ أَمْ لَمْ تُنذِرْهُمْ لَا يُؤْمِنُونَ ﴿١٠﴾

    البقرة ٦

    إِنَّ ٱلَّذِينَ كَفَرُوا۟ سَوَآءٌ عَلَيْهِمْ ءَأَنذَرْتَهُمْ أَمْ لَمْ تُنذِرْهُمْ لَا يُؤْمِنُونَ ﴿٦﴾

    آخرها يشبه آية هذه السورة: «سَوَآءٌ» بدل «وَسَوَآءٌ»

  3. وَمَا عَلَيْنَآ إِلَّا ٱلْبَلَـٰغُ ٱلْمُبِينُ ﴿١٧﴾

    النور ٥٤

    قُلْ أَطِيعُوا۟ ٱللَّهَ وَأَطِيعُوا۟ ٱلرَّسُولَ ۖ فَإِن تَوَلَّوْا۟ فَإِنَّمَا عَلَيْهِ مَا حُمِّلَ وَعَلَيْكُم مَّا حُمِّلْتُمْ ۖ وَإِن تُطِيعُوهُ تَهْتَدُوا۟ ۚ وَمَا عَلَى ٱلرَّسُولِ إِلَّا ٱلْبَلَـٰغُ ٱلْمُبِينُ ﴿٥٤﴾

    آخرها يشبه آية هذه السورة: «عَلَى ٱلرَّسُولِ» بدل «عَلَيْنَآ»

    العنكبوت ١٨

    وَإِن تُكَذِّبُوا۟ فَقَدْ كَذَّبَ أُمَمٌ مِّن قَبْلِكُمْ ۖ وَمَا عَلَى ٱلرَّسُولِ إِلَّا ٱلْبَلَـٰغُ ٱلْمُبِينُ ﴿١٨﴾

    آخرها يشبه آية هذه السورة: «عَلَى ٱلرَّسُولِ» بدل «عَلَيْنَآ»

  4. يَـٰحَسْرَةً عَلَى ٱلْعِبَادِ ۚ مَا يَأْتِيهِم مِّن رَّسُولٍ إِلَّا كَانُوا۟ بِهِۦ يَسْتَهْزِءُونَ ﴿٣٠﴾

    الحجر ١١

    وَمَا يَأْتِيهِم مِّن رَّسُولٍ إِلَّا كَانُوا۟ بِهِۦ يَسْتَهْزِءُونَ ﴿١١﴾

    «وَمَا» بدل «يَـٰحَسْرَةً عَلَى ٱلْعِبَادِ مَا»

  5. وَيَقُولُونَ مَتَىٰ هَـٰذَا ٱلْوَعْدُ إِن كُنتُمْ صَـٰدِقِينَ ﴿٤٨﴾

    السجدة ٢٨

    وَيَقُولُونَ مَتَىٰ هَـٰذَا ٱلْفَتْحُ إِن كُنتُمْ صَـٰدِقِينَ ﴿٢٨﴾

    «ٱلْفَتْحُ» بدل «ٱلْوَعْدُ»

  6. فَٱلْيَوْمَ لَا تُظْلَمُ نَفْسٌ شَيْـًٔا وَلَا تُجْزَوْنَ إِلَّا مَا كُنتُمْ تَعْمَلُونَ ﴿٥٤﴾

    الصافات ٣٩

    وَمَا تُجْزَوْنَ إِلَّا مَا كُنتُمْ تَعْمَلُونَ ﴿٣٩﴾

    «وَمَا» بدل «فَٱلْيَوْمَ لَا تُظْلَمُ نَفْسٌ شَيْـًٔا وَلَا»

  7. وَأَنِ ٱعْبُدُونِى ۚ هَـٰذَا صِرَٰطٌ مُّسْتَقِيمٌ ﴿٦١﴾

    الحجر ٤١

    قَالَ هَـٰذَا صِرَٰطٌ عَلَىَّ مُسْتَقِيمٌ ﴿٤١﴾

    «قَالَ» بدل «وَأَنِ ٱعْبُدُونِى»؛ زيادة «عَلَىَّ»

  8. وَمَا عَلَّمْنَـٰهُ ٱلشِّعْرَ وَمَا يَنۢبَغِى لَهُۥٓ ۚ إِنْ هُوَ إِلَّا ذِكْرٌ وَقُرْءَانٌ مُّبِينٌ ﴿٦٩﴾

    ص ٨٧؛ التكوير ٢٧

    إِنْ هُوَ إِلَّا ذِكْرٌ لِّلْعَـٰلَمِينَ ﴿٨٧﴾

    تشبه آخرَ آية هذه السورة: «لِّلْعَـٰلَمِينَ» بدل «وَقُرْءَانٌ مُّبِينٌ»

  9. وَٱتَّخَذُوا۟ مِن دُونِ ٱللَّهِ ءَالِهَةً لَّعَلَّهُمْ يُنصَرُونَ ﴿٧٤﴾

    مريم ٨١

    وَٱتَّخَذُوا۟ مِن دُونِ ٱللَّهِ ءَالِهَةً لِّيَكُونُوا۟ لَهُمْ عِزًّا ﴿٨١﴾

    «لِّيَكُونُوا۟ لَهُمْ عِزًّا» بدل «لَّعَلَّهُمْ يُنصَرُونَ»

    الأحقاف ٢٨

    فَلَوْلَا نَصَرَهُمُ ٱلَّذِينَ ٱتَّخَذُوا۟ مِن دُونِ ٱللَّهِ قُرْبَانًا ءَالِهَةًۢ ۖ بَلْ ضَلُّوا۟ عَنْهُمْ ۚ وَذَٰلِكَ إِفْكُهُمْ وَمَا كَانُوا۟ يَفْتَرُونَ ﴿٢٨﴾

    «فَلَوْلَا نَصَرَهُمُ ٱلَّذِينَ ٱتَّخَذُوا۟» بدل «وَٱتَّخَذُوا۟»؛ «قُرْبَانًا ءَالِهَةًۢ بَلْ … كَانُوا۟ يَفْتَرُونَ» بدل «ءَالِهَةً لَّعَلَّهُمْ يُنصَرُونَ»

  10. أَوَلَمْ يَرَ ٱلْإِنسَـٰنُ أَنَّا خَلَقْنَـٰهُ مِن نُّطْفَةٍ فَإِذَا هُوَ خَصِيمٌ مُّبِينٌ ﴿٧٧﴾

    النحل ٤

    خَلَقَ ٱلْإِنسَـٰنَ مِن نُّطْفَةٍ فَإِذَا هُوَ خَصِيمٌ مُّبِينٌ ﴿٤﴾

    «خَلَقَ ٱلْإِنسَـٰنَ» بدل «أَوَلَمْ يَرَ ٱلْإِنسَـٰنُ أَنَّا خَلَقْنَـٰهُ»

Aleyas similares en otras suras

  1. تَنزِيلَ ٱلْعَزِيزِ ٱلرَّحِيمِ ﴿٥﴾

    فصلت 2

    تَنزِيلٌ مِّنَ ٱلرَّحْمَـٰنِ ٱلرَّحِيمِ ﴿٢﴾

    «مِّنَ ٱلرَّحْمَـٰنِ» en lugar de «ٱلْعَزِيزِ»

  2. وَسَوَآءٌ عَلَيْهِمْ ءَأَنذَرْتَهُمْ أَمْ لَمْ تُنذِرْهُمْ لَا يُؤْمِنُونَ ﴿١٠﴾

    البقرة 6

    إِنَّ ٱلَّذِينَ كَفَرُوا۟ سَوَآءٌ عَلَيْهِمْ ءَأَنذَرْتَهُمْ أَمْ لَمْ تُنذِرْهُمْ لَا يُؤْمِنُونَ ﴿٦﴾

    Su final se parece a la aleya de esta sura: «سَوَآءٌ» en lugar de «وَسَوَآءٌ»

  3. وَمَا عَلَيْنَآ إِلَّا ٱلْبَلَـٰغُ ٱلْمُبِينُ ﴿١٧﴾

    النور 54

    قُلْ أَطِيعُوا۟ ٱللَّهَ وَأَطِيعُوا۟ ٱلرَّسُولَ ۖ فَإِن تَوَلَّوْا۟ فَإِنَّمَا عَلَيْهِ مَا حُمِّلَ وَعَلَيْكُم مَّا حُمِّلْتُمْ ۖ وَإِن تُطِيعُوهُ تَهْتَدُوا۟ ۚ وَمَا عَلَى ٱلرَّسُولِ إِلَّا ٱلْبَلَـٰغُ ٱلْمُبِينُ ﴿٥٤﴾

    Su final se parece a la aleya de esta sura: «عَلَى ٱلرَّسُولِ» en lugar de «عَلَيْنَآ»

    العنكبوت 18

    وَإِن تُكَذِّبُوا۟ فَقَدْ كَذَّبَ أُمَمٌ مِّن قَبْلِكُمْ ۖ وَمَا عَلَى ٱلرَّسُولِ إِلَّا ٱلْبَلَـٰغُ ٱلْمُبِينُ ﴿١٨﴾

    Su final se parece a la aleya de esta sura: «عَلَى ٱلرَّسُولِ» en lugar de «عَلَيْنَآ»

  4. يَـٰحَسْرَةً عَلَى ٱلْعِبَادِ ۚ مَا يَأْتِيهِم مِّن رَّسُولٍ إِلَّا كَانُوا۟ بِهِۦ يَسْتَهْزِءُونَ ﴿٣٠﴾

    الحجر 11

    وَمَا يَأْتِيهِم مِّن رَّسُولٍ إِلَّا كَانُوا۟ بِهِۦ يَسْتَهْزِءُونَ ﴿١١﴾

    «وَمَا» en lugar de «يَـٰحَسْرَةً عَلَى ٱلْعِبَادِ مَا»

  5. وَيَقُولُونَ مَتَىٰ هَـٰذَا ٱلْوَعْدُ إِن كُنتُمْ صَـٰدِقِينَ ﴿٤٨﴾

    السجدة 28

    وَيَقُولُونَ مَتَىٰ هَـٰذَا ٱلْفَتْحُ إِن كُنتُمْ صَـٰدِقِينَ ﴿٢٨﴾

    «ٱلْفَتْحُ» en lugar de «ٱلْوَعْدُ»

  6. فَٱلْيَوْمَ لَا تُظْلَمُ نَفْسٌ شَيْـًٔا وَلَا تُجْزَوْنَ إِلَّا مَا كُنتُمْ تَعْمَلُونَ ﴿٥٤﴾

    الصافات 39

    وَمَا تُجْزَوْنَ إِلَّا مَا كُنتُمْ تَعْمَلُونَ ﴿٣٩﴾

    «وَمَا» en lugar de «فَٱلْيَوْمَ لَا تُظْلَمُ نَفْسٌ شَيْـًٔا وَلَا»

  7. وَأَنِ ٱعْبُدُونِى ۚ هَـٰذَا صِرَٰطٌ مُّسْتَقِيمٌ ﴿٦١﴾

    الحجر 41

    قَالَ هَـٰذَا صِرَٰطٌ عَلَىَّ مُسْتَقِيمٌ ﴿٤١﴾

    «قَالَ» en lugar de «وَأَنِ ٱعْبُدُونِى»; se añade «عَلَىَّ»

  8. وَمَا عَلَّمْنَـٰهُ ٱلشِّعْرَ وَمَا يَنۢبَغِى لَهُۥٓ ۚ إِنْ هُوَ إِلَّا ذِكْرٌ وَقُرْءَانٌ مُّبِينٌ ﴿٦٩﴾

    ص 87; التكوير 27

    إِنْ هُوَ إِلَّا ذِكْرٌ لِّلْعَـٰلَمِينَ ﴿٨٧﴾

    Se parece al final de la aleya de esta sura: «لِّلْعَـٰلَمِينَ» en lugar de «وَقُرْءَانٌ مُّبِينٌ»

  9. وَٱتَّخَذُوا۟ مِن دُونِ ٱللَّهِ ءَالِهَةً لَّعَلَّهُمْ يُنصَرُونَ ﴿٧٤﴾

    مريم 81

    وَٱتَّخَذُوا۟ مِن دُونِ ٱللَّهِ ءَالِهَةً لِّيَكُونُوا۟ لَهُمْ عِزًّا ﴿٨١﴾

    «لِّيَكُونُوا۟ لَهُمْ عِزًّا» en lugar de «لَّعَلَّهُمْ يُنصَرُونَ»

    الأحقاف 28

    فَلَوْلَا نَصَرَهُمُ ٱلَّذِينَ ٱتَّخَذُوا۟ مِن دُونِ ٱللَّهِ قُرْبَانًا ءَالِهَةًۢ ۖ بَلْ ضَلُّوا۟ عَنْهُمْ ۚ وَذَٰلِكَ إِفْكُهُمْ وَمَا كَانُوا۟ يَفْتَرُونَ ﴿٢٨﴾

    «فَلَوْلَا نَصَرَهُمُ ٱلَّذِينَ ٱتَّخَذُوا۟» en lugar de «وَٱتَّخَذُوا۟»; «قُرْبَانًا ءَالِهَةًۢ بَلْ … كَانُوا۟ يَفْتَرُونَ» en lugar de «ءَالِهَةً لَّعَلَّهُمْ يُنصَرُونَ»

  10. أَوَلَمْ يَرَ ٱلْإِنسَـٰنُ أَنَّا خَلَقْنَـٰهُ مِن نُّطْفَةٍ فَإِذَا هُوَ خَصِيمٌ مُّبِينٌ ﴿٧٧﴾

    النحل 4

    خَلَقَ ٱلْإِنسَـٰنَ مِن نُّطْفَةٍ فَإِذَا هُوَ خَصِيمٌ مُّبِينٌ ﴿٤﴾

    «خَلَقَ ٱلْإِنسَـٰنَ» en lugar de «أَوَلَمْ يَرَ ٱلْإِنسَـٰنُ أَنَّا خَلَقْنَـٰهُ»

«من يحيي العظام وهي رميم»: «قل يحييها الذي أنشأها أوّل مرّة»، الشجر الأخضر، «كن فيكون»

أَوَلَمْ يَرَ ٱلْإِنسَـٰنُ أَنَّا خَلَقْنَـٰهُ مِن نُّطْفَةٍ فَإِذَا هُوَ خَصِيمٌ مُّبِينٌ ﴿٧٧﴾ وَضَرَبَ لَنَا مَثَلًا وَنَسِىَ خَلْقَهُۥ ۖ قَالَ مَن يُحْىِ ٱلْعِظَـٰمَ وَهِىَ رَمِيمٌ ﴿٧٨﴾ قُلْ يُحْيِيهَا ٱلَّذِىٓ أَنشَأَهَآ أَوَّلَ مَرَّةٍ ۖ وَهُوَ بِكُلِّ خَلْقٍ عَلِيمٌ ﴿٧٩﴾ ٱلَّذِى جَعَلَ لَكُم مِّنَ ٱلشَّجَرِ ٱلْأَخْضَرِ نَارًا فَإِذَآ أَنتُم مِّنْهُ تُوقِدُونَ ﴿٨٠﴾ أَوَلَيْسَ ٱلَّذِى خَلَقَ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضَ بِقَـٰدِرٍ عَلَىٰٓ أَن يَخْلُقَ مِثْلَهُم ۚ بَلَىٰ وَهُوَ ٱلْخَلَّـٰقُ ٱلْعَلِيمُ ﴿٨١﴾ إِنَّمَآ أَمْرُهُۥٓ إِذَآ أَرَادَ شَيْـًٔا أَن يَقُولَ لَهُۥ كُن فَيَكُونُ ﴿٨٢﴾ فَسُبْحَـٰنَ ٱلَّذِى بِيَدِهِۦ مَلَكُوتُ كُلِّ شَىْءٍ وَإِلَيْهِ تُرْجَعُونَ ﴿٨٣﴾
