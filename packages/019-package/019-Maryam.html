  1. ١ آيتان تشتركان في «أَنَّىٰ يَكُونُ لِى غُلَـٰمٌ»

    قَالَ رَبِّ أَنَّىٰ يَكُونُ لِى غُلَـٰمٌ وَكَانَتِ ٱمْرَأَتِى عَاقِرًا وَقَدْ بَلَغْتُ مِنَ ٱلْكِبَرِ عِتِيًّا ﴿٨﴾

    قَالَتْ أَنَّىٰ يَكُونُ لِى غُلَـٰمٌ وَلَمْ يَمْسَسْنِى بَشَرٌ وَلَمْ أَكُ بَغِيًّا ﴿٢٠﴾

    قبلها «قَالَتْ» بدل «رَبِّ»؛ بعدها «وَلَمْ» بدل «وَكَانَتِ»

  2. ٢ آيتان تشتركان في «قَالَ كَذَٰلِكَ قَالَ رَبُّكَ هُوَ عَلَىَّ هَيِّنٌ»

    قَالَ كَذَٰلِكَ قَالَ رَبُّكَ هُوَ عَلَىَّ هَيِّنٌ وَقَدْ خَلَقْتُكَ مِن قَبْلُ وَلَمْ تَكُ شَيْـًٔا ﴿٩﴾

    قَالَ كَذَٰلِكِ قَالَ رَبُّكِ هُوَ عَلَىَّ هَيِّنٌ ۖ وَلِنَجْعَلَهُۥٓ ءَايَةً لِّلنَّاسِ وَرَحْمَةً مِّنَّا ۚ وَكَانَ أَمْرًا مَّقْضِيًّا ﴿٢١﴾

    بعدها «وَلِنَجْعَلَهُۥٓ» بدل «وَقَدْ»

  3. ٣ آيتان تشتركان في «وَكَمْ أَهْلَكْنَا قَبْلَهُم مِّن قَرْنٍ»

    وَكَمْ أَهْلَكْنَا قَبْلَهُم مِّن قَرْنٍ هُمْ أَحْسَنُ أَثَـٰثًا وَرِءْيًا ﴿٧٤﴾

    وَكَمْ أَهْلَكْنَا قَبْلَهُم مِّن قَرْنٍ هَلْ تُحِسُّ مِنْهُم مِّنْ أَحَدٍ أَوْ تَسْمَعُ لَهُمْ رِكْزًۢا ﴿٩٨﴾

    بعدها «هَلْ» بدل «هُمْ»

  4. ٤ آيتان تشتركان في «ٱتَّخَذَ عِندَ ٱلرَّحْمَـٰنِ عَهْدًا»

    أَطَّلَعَ ٱلْغَيْبَ أَمِ ٱتَّخَذَ عِندَ ٱلرَّحْمَـٰنِ عَهْدًا ﴿٧٨﴾

    لَّا يَمْلِكُونَ ٱلشَّفَـٰعَةَ إِلَّا مَنِ ٱتَّخَذَ عِندَ ٱلرَّحْمَـٰنِ عَهْدًا ﴿٨٧﴾

    قبلها «مَنِ» بدل «أَمِ»

  1. قَالَ رَبِّ ٱجْعَل لِّىٓ ءَايَةً ۚ قَالَ ءَايَتُكَ أَلَّا تُكَلِّمَ ٱلنَّاسَ ثَلَـٰثَ لَيَالٍ سَوِيًّا ﴿١٠﴾

    آل عمران ٤١

    قَالَ رَبِّ ٱجْعَل لِّىٓ ءَايَةً ۖ قَالَ ءَايَتُكَ أَلَّا تُكَلِّمَ ٱلنَّاسَ ثَلَـٰثَةَ أَيَّامٍ إِلَّا رَمْزًا ۗ وَٱذْكُر رَّبَّكَ كَثِيرًا وَسَبِّحْ بِٱلْعَشِىِّ وَٱلْإِبْكَـٰرِ ﴿٤١﴾

    «ثَلَـٰثَةَ أَيَّامٍ إِلَّا … بِٱلْعَشِىِّ وَٱلْإِبْكَـٰرِ» بدل «ثَلَـٰثَ لَيَالٍ سَوِيًّا»

  2. وَإِنَّ ٱللَّهَ رَبِّى وَرَبُّكُمْ فَٱعْبُدُوهُ ۚ هَـٰذَا صِرَٰطٌ مُّسْتَقِيمٌ ﴿٣٦﴾

    آل عمران ٥١

    إِنَّ ٱللَّهَ رَبِّى وَرَبُّكُمْ فَٱعْبُدُوهُ ۗ هَـٰذَا صِرَٰطٌ مُّسْتَقِيمٌ ﴿٥١﴾

    «إِنَّ» بدل «وَإِنَّ»

    الزخرف ٦٤

    إِنَّ ٱللَّهَ هُوَ رَبِّى وَرَبُّكُمْ فَٱعْبُدُوهُ ۚ هَـٰذَا صِرَٰطٌ مُّسْتَقِيمٌ ﴿٦٤﴾

    «إِنَّ ٱللَّهَ هُوَ» بدل «وَإِنَّ ٱللَّهَ»

  3. فَٱخْتَلَفَ ٱلْأَحْزَابُ مِنۢ بَيْنِهِمْ ۖ فَوَيْلٌ لِّلَّذِينَ كَفَرُوا۟ مِن مَّشْهَدِ يَوْمٍ عَظِيمٍ ﴿٣٧﴾

    الزخرف ٦٥

    فَٱخْتَلَفَ ٱلْأَحْزَابُ مِنۢ بَيْنِهِمْ ۖ فَوَيْلٌ لِّلَّذِينَ ظَلَمُوا۟ مِنْ عَذَابِ يَوْمٍ أَلِيمٍ ﴿٦٥﴾

    «ظَلَمُوا۟» بدل «كَفَرُوا۟»؛ «عَذَابِ» بدل «مَّشْهَدِ»؛ «أَلِيمٍ» بدل «عَظِيمٍ»

  4. لَّا يَسْمَعُونَ فِيهَا لَغْوًا إِلَّا سَلَـٰمًا ۖ وَلَهُمْ رِزْقُهُمْ فِيهَا بُكْرَةً وَعَشِيًّا ﴿٦٢﴾

    الواقعة ٢٥

    لَا يَسْمَعُونَ فِيهَا لَغْوًا وَلَا تَأْثِيمًا ﴿٢٥﴾

    «وَلَا تَأْثِيمًا» بدل «إِلَّا سَلَـٰمًا وَلَهُمْ رِزْقُهُمْ فِيهَا بُكْرَةً وَعَشِيًّا»

    النبإ ٣٥

    لَّا يَسْمَعُونَ فِيهَا لَغْوًا وَلَا كِذَّٰبًا ﴿٣٥﴾

    «وَلَا كِذَّٰبًا» بدل «إِلَّا سَلَـٰمًا وَلَهُمْ رِزْقُهُمْ فِيهَا بُكْرَةً وَعَشِيًّا»

  5. رَّبُّ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ وَمَا بَيْنَهُمَا فَٱعْبُدْهُ وَٱصْطَبِرْ لِعِبَـٰدَتِهِۦ ۚ هَلْ تَعْلَمُ لَهُۥ سَمِيًّا ﴿٦٥﴾

    الصافات ٥

    رَّبُّ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ وَمَا بَيْنَهُمَا وَرَبُّ ٱلْمَشَـٰرِقِ ﴿٥﴾

    «وَرَبُّ ٱلْمَشَـٰرِقِ» بدل «فَٱعْبُدْهُ وَٱصْطَبِرْ لِعِبَـٰدَتِهِۦ هَلْ تَعْلَمُ لَهُۥ سَمِيًّا»

    ص ٦٦

    رَبُّ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ وَمَا بَيْنَهُمَا ٱلْعَزِيزُ ٱلْغَفَّـٰرُ ﴿٦٦﴾

    «ٱلْعَزِيزُ ٱلْغَفَّـٰرُ» بدل «فَٱعْبُدْهُ وَٱصْطَبِرْ لِعِبَـٰدَتِهِۦ هَلْ تَعْلَمُ لَهُۥ سَمِيًّا»

  6. وَكَمْ أَهْلَكْنَا قَبْلَهُم مِّن قَرْنٍ هُمْ أَحْسَنُ أَثَـٰثًا وَرِءْيًا ﴿٧٤﴾

    ق ٣٦

    وَكَمْ أَهْلَكْنَا قَبْلَهُم مِّن قَرْنٍ هُمْ أَشَدُّ مِنْهُم بَطْشًا فَنَقَّبُوا۟ فِى ٱلْبِلَـٰدِ هَلْ مِن مَّحِيصٍ ﴿٣٦﴾

    «أَشَدُّ مِنْهُم بَطْشًا … مِن مَّحِيصٍ» بدل «أَحْسَنُ أَثَـٰثًا وَرِءْيًا»

  7. وَٱتَّخَذُوا۟ مِن دُونِ ٱللَّهِ ءَالِهَةً لِّيَكُونُوا۟ لَهُمْ عِزًّا ﴿٨١﴾

«أفرأيت الذي كفر بآياتنا وقال لأوتينّ مالاً وولداً»: «أطّلع الغيب أم اتّخذ عند الرحمن عهداً»، «كلّا سنكتب ما يقول»، «ألم تر أنّا أرسلنا الشياطين على الكافرين تؤزّهم أزّاً»، و«يوم نحشر المتّقين إلى الرحمن وفداً»

أَفَرَءَيْتَ ٱلَّذِى كَفَرَ بِـَٔايَـٰتِنَا وَقَالَ لَأُوتَيَنَّ مَالًا وَوَلَدًا ﴿٧٧﴾ أَطَّلَعَ ٱلْغَيْبَ أَمِ ٱتَّخَذَ عِندَ ٱلرَّحْمَـٰنِ عَهْدًا ﴿٧٨﴾ كَلَّا ۚ سَنَكْتُبُ مَا يَقُولُ وَنَمُدُّ لَهُۥ مِنَ ٱلْعَذَابِ مَدًّا ﴿٧٩﴾ وَنَرِثُهُۥ مَا يَقُولُ وَيَأْتِينَا فَرْدًا ﴿٨٠﴾ وَٱتَّخَذُوا۟ مِن دُونِ ٱللَّهِ ءَالِهَةً لِّيَكُونُوا۟ لَهُمْ عِزًّا ﴿٨١﴾ كَلَّا ۚ سَيَكْفُرُونَ بِعِبَادَتِهِمْ وَيَكُونُونَ عَلَيْهِمْ ضِدًّا ﴿٨٢﴾ أَلَمْ تَرَ أَنَّآ أَرْسَلْنَا ٱلشَّيَـٰطِينَ عَلَى ٱلْكَـٰفِرِينَ تَؤُزُّهُمْ أَزًّا ﴿٨٣﴾ فَلَا تَعْجَلْ عَلَيْهِمْ ۖ إِنَّمَا نَعُدُّ لَهُمْ عَدًّا ﴿٨٤﴾ يَوْمَ نَحْشُرُ ٱلْمُتَّقِينَ إِلَى ٱلرَّحْمَـٰنِ وَفْدًا ﴿٨٥﴾ وَنَسُوقُ ٱلْمُجْرِمِينَ إِلَىٰ جَهَنَّمَ وِرْدًا ﴿٨٦﴾ لَّا يَمْلِكُونَ ٱلشَّفَـٰعَةَ إِلَّا مَنِ ٱتَّخَذَ عِندَ ٱلرَّحْمَـٰنِ عَهْدًا ﴿٨٧﴾

«وقالوا اتّخذ الرحمن ولداً»: «لقد جئتم شيئاً إدّاً»، «تكاد السماوات يتفطّرن منه»، «إن كلّ من في السماوات والأرض إلّا آتي الرحمن عبداً»، «سيجعل لهم الرحمن ودّاً»، و«هل تحسّ منهم من أحد أو تسمع لهم ركزاً»

وَقَالُوا۟ ٱتَّخَذَ ٱلرَّحْمَـٰنُ وَلَدًا ﴿٨٨﴾ لَّقَدْ جِئْتُمْ شَيْـًٔا إِدًّا ﴿٨٩﴾ تَكَادُ ٱلسَّمَـٰوَٰتُ يَتَفَطَّرْنَ مِنْهُ وَتَنشَقُّ ٱلْأَرْضُ وَتَخِرُّ ٱلْجِبَالُ هَدًّا ﴿٩٠﴾ أَن دَعَوْا۟ لِلرَّحْمَـٰنِ وَلَدًا ﴿٩١﴾ وَمَا يَنۢبَغِى لِلرَّحْمَـٰنِ أَن يَتَّخِذَ وَلَدًا ﴿٩٢﴾ إِن كُلُّ مَن فِى ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ إِلَّآ ءَاتِى ٱلرَّحْمَـٰنِ عَبْدًا ﴿٩٣﴾ لَّقَدْ أَحْصَىٰهُمْ وَعَدَّهُمْ عَدًّا ﴿٩٤﴾ وَكُلُّهُمْ ءَاتِيهِ يَوْمَ ٱلْقِيَـٰمَةِ فَرْدًا ﴿٩٥﴾ إِنَّ ٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّـٰلِحَـٰتِ سَيَجْعَلُ لَهُمُ ٱلرَّحْمَـٰنُ وُدًّا ﴿٩٦﴾ فَإِنَّمَا يَسَّرْنَـٰهُ بِلِسَانِكَ لِتُبَشِّرَ بِهِ ٱلْمُتَّقِينَ وَتُنذِرَ بِهِۦ قَوْمًا لُّدًّا ﴿٩٧﴾ وَكَمْ أَهْلَكْنَا قَبْلَهُم مِّن قَرْنٍ هَلْ تُحِسُّ مِنْهُم مِّنْ أَحَدٍ أَوْ تَسْمَعُ لَهُمْ رِكْزًۢا ﴿٩٨﴾
