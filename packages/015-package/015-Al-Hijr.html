    ٱللَّهُ يَتَوَفَّى ٱلْأَنفُسَ حِينَ مَوْتِهَا وَٱلَّتِى لَمْ تَمُتْ فِى مَنَامِهَا ۖ فَيُمْسِكُ ٱلَّتِى قَضَىٰ عَلَيْهَا ٱلْمَوْتَ وَيُرْسِلُ ٱلْأُخْرَىٰٓ إِلَىٰٓ أَجَلٍ مُّسَمًّى ۚ إِنَّ فِى ذَٰلِكَ لَـَٔايَـٰتٍ لِّقَوْمٍ يَتَفَكَّرُونَ ﴿٤٢﴾

    آخرها يشبه آية هذه السورة: «لِّقَوْمٍ يَتَفَكَّرُونَ» بدل «لِّلْمُتَوَسِّمِينَ»

    الزمر ٥٢

    أَوَلَمْ يَعْلَمُوٓا۟ أَنَّ ٱللَّهَ يَبْسُطُ ٱلرِّزْقَ لِمَن يَشَآءُ وَيَقْدِرُ ۚ إِنَّ فِى ذَٰلِكَ لَـَٔايَـٰتٍ لِّقَوْمٍ يُؤْمِنُونَ ﴿٥٢﴾

    آخرها يشبه آية هذه السورة: «لِّقَوْمٍ يُؤْمِنُونَ» بدل «لِّلْمُتَوَسِّمِينَ»

    الشورى ٣٣

    إِن يَشَأْ يُسْكِنِ ٱلرِّيحَ فَيَظْلَلْنَ رَوَاكِدَ عَلَىٰ ظَهْرِهِۦٓ ۚ إِنَّ فِى ذَٰلِكَ لَـَٔايَـٰتٍ لِّكُلِّ صَبَّارٍ شَكُورٍ ﴿٣٣﴾

    آخرها يشبه آية هذه السورة: «لِّكُلِّ صَبَّارٍ شَكُورٍ» بدل «لِّلْمُتَوَسِّمِينَ»

    الجاثية ١٣

    وَسَخَّرَ لَكُم مَّا فِى ٱلسَّمَـٰوَٰتِ وَمَا فِى ٱلْأَرْضِ جَمِيعًا مِّنْهُ ۚ إِنَّ فِى ذَٰلِكَ لَـَٔايَـٰتٍ لِّقَوْمٍ يَتَفَكَّرُونَ ﴿١٣﴾

    آخرها يشبه آية هذه السورة: «لِّقَوْمٍ يَتَفَكَّرُونَ» بدل «لِّلْمُتَوَسِّمِينَ»

  21. إِنَّ فِى ذَٰلِكَ لَـَٔايَةً لِّلْمُؤْمِنِينَ ﴿٧٧﴾

    البقرة ٢٤٨

    وَقَالَ لَهُمْ نَبِيُّهُمْ إِنَّ ءَايَةَ مُلْكِهِۦٓ أَن يَأْتِيَكُمُ ٱلتَّابُوتُ فِيهِ سَكِينَةٌ مِّن رَّبِّكُمْ وَبَقِيَّةٌ مِّمَّا تَرَكَ ءَالُ مُوسَىٰ وَءَالُ هَـٰرُونَ تَحْمِلُهُ ٱلْمَلَـٰٓئِكَةُ ۚ إِنَّ فِى ذَٰلِكَ لَـَٔايَةً لَّكُمْ إِن كُنتُم مُّؤْمِنِينَ ﴿٢٤٨﴾

    زيادة «وَقَالَ لَهُمْ نَبِيُّهُمْ … تَحْمِلُهُ ٱلْمَلَـٰٓئِكَةُ» في أولها؛ «لَّكُمْ إِن كُنتُم مُّؤْمِنِينَ» بدل «لِّلْمُؤْمِنِينَ»

    آل عمران ٤٩

    وَرَسُولًا إِلَىٰ بَنِىٓ إِسْرَٰٓءِيلَ أَنِّى قَدْ جِئْتُكُم بِـَٔايَةٍ مِّن رَّبِّكُمْ ۖ أَنِّىٓ أَخْلُقُ لَكُم مِّنَ ٱلطِّينِ كَهَيْـَٔةِ ٱلطَّيْرِ فَأَنفُخُ فِيهِ فَيَكُونُ طَيْرًۢا بِإِذْنِ ٱللَّهِ ۖ وَأُبْرِئُ ٱلْأَكْمَهَ وَٱلْأَبْرَصَ وَأُحْىِ ٱلْمَوْتَىٰ بِإِذْنِ ٱللَّهِ ۖ وَأُنَبِّئُكُم بِمَا تَأْكُلُونَ وَمَا تَدَّخِرُونَ فِى بُيُوتِكُمْ ۚ إِنَّ فِى ذَٰلِكَ لَـَٔايَةً لَّكُمْ إِن كُنتُم مُّؤْمِنِينَ ﴿٤٩﴾

    زيادة «وَرَسُولًا إِلَىٰ بَنِىٓ … فِى بُيُوتِكُمْ» في أولها؛ «لَّكُمْ إِن كُنتُم مُّؤْمِنِينَ» بدل «لِّلْمُؤْمِنِينَ»

    هود ١٠٣

    إِنَّ فِى ذَٰلِكَ لَـَٔايَةً لِّمَنْ خَافَ عَذَابَ ٱلْـَٔاخِرَةِ ۚ ذَٰلِكَ يَوْمٌ مَّجْمُوعٌ لَّهُ ٱلنَّاسُ وَذَٰلِكَ يَوْمٌ مَّشْهُودٌ ﴿١٠٣﴾

    «لِّمَنْ خَافَ عَذَابَ … يَوْمٌ مَّشْهُودٌ» بدل «لِّلْمُؤْمِنِينَ»

    النحل ١١

    يُنۢبِتُ لَكُم بِهِ ٱلزَّرْعَ وَٱلزَّيْتُونَ وَٱلنَّخِيلَ وَٱلْأَعْنَـٰبَ وَمِن كُلِّ ٱلثَّمَرَٰتِ ۗ إِنَّ فِى ذَٰلِكَ لَـَٔايَةً لِّقَوْمٍ يَتَفَكَّرُونَ ﴿١١﴾

    آخرها يشبه آية هذه السورة: «لِّقَوْمٍ يَتَفَكَّرُونَ» بدل «لِّلْمُؤْمِنِينَ»

    النحل ١٣

    وَمَا ذَرَأَ لَكُمْ فِى ٱلْأَرْضِ مُخْتَلِفًا أَلْوَٰنُهُۥٓ ۗ إِنَّ فِى ذَٰلِكَ لَـَٔايَةً لِّقَوْمٍ يَذَّكَّرُونَ ﴿١٣﴾

    آخرها يشبه آية هذه السورة: «لِّقَوْمٍ يَذَّكَّرُونَ» بدل «لِّلْمُؤْمِنِينَ»

    النحل ٦٥

    وَٱللَّهُ أَنزَلَ مِنَ ٱلسَّمَآءِ مَآءً فَأَحْيَا بِهِ ٱلْأَرْضَ بَعْدَ مَوْتِهَآ ۚ إِنَّ فِى ذَٰلِكَ لَـَٔايَةً لِّقَوْمٍ يَسْمَعُونَ ﴿٦٥﴾

    آخرها يشبه آية هذه السورة: «لِّقَوْمٍ يَسْمَعُونَ» بدل «لِّلْمُؤْمِنِينَ»

    النحل ٦٧

    وَمِن ثَمَرَٰتِ ٱلنَّخِيلِ وَٱلْأَعْنَـٰبِ تَتَّخِذُونَ مِنْهُ سَكَرًا وَرِزْقًا حَسَنًا ۗ إِنَّ فِى ذَٰلِكَ لَـَٔايَةً لِّقَوْمٍ يَعْقِلُونَ ﴿٦٧﴾

    آخرها يشبه آية هذه السورة: «لِّقَوْمٍ يَعْقِلُونَ» بدل «لِّلْمُؤْمِنِينَ»

    النحل ٦٩

    ثُمَّ كُلِى مِن كُلِّ ٱلثَّمَرَٰتِ فَٱسْلُكِى سُبُلَ رَبِّكِ ذُلُلًا ۚ يَخْرُجُ مِنۢ بُطُونِهَا شَرَابٌ مُّخْتَلِفٌ أَلْوَٰنُهُۥ فِيهِ شِفَآءٌ لِّلنَّاسِ ۗ إِنَّ فِى ذَٰلِكَ لَـَٔايَةً لِّقَوْمٍ يَتَفَكَّرُونَ ﴿٦٩﴾

    آخرها يشبه آية هذه السورة: «لِّقَوْمٍ يَتَفَكَّرُونَ» بدل «لِّلْمُؤْمِنِينَ»

    الشعراء ٨، ٦٧، ١٠٣، ١٢١، ١٧٤، ١٩٠

    إِنَّ فِى ذَٰلِكَ لَـَٔايَةً ۖ وَمَا كَانَ أَكْثَرُهُم مُّؤْمِنِينَ ﴿٨﴾

    «وَمَا كَانَ أَكْثَرُهُم مُّؤْمِنِينَ» بدل «لِّلْمُؤْمِنِينَ»

    الشعراء ١٣٩

    فَكَذَّبُوهُ فَأَهْلَكْنَـٰهُمْ ۗ إِنَّ فِى ذَٰلِكَ لَـَٔايَةً ۖ وَمَا كَانَ أَكْثَرُهُم مُّؤْمِنِينَ ﴿١٣٩﴾

    زيادة «فَكَذَّبُوهُ فَأَهْلَكْنَـٰهُمْ» في أولها؛ «وَمَا كَانَ أَكْثَرُهُم مُّؤْمِنِينَ» بدل «لِّلْمُؤْمِنِينَ»

    الشعراء ١٥٨

    فَأَخَذَهُمُ ٱلْعَذَابُ ۗ إِنَّ فِى ذَٰلِكَ لَـَٔايَةً ۖ وَمَا كَانَ أَكْثَرُهُم مُّؤْمِنِينَ ﴿١٥٨﴾

    زيادة «فَأَخَذَهُمُ ٱلْعَذَابُ» في أولها؛ «وَمَا كَانَ أَكْثَرُهُم مُّؤْمِنِينَ» بدل «لِّلْمُؤْمِنِينَ»

    النمل ٥٢

    فَتِلْكَ بُيُوتُهُمْ خَاوِيَةًۢ بِمَا ظَلَمُوٓا۟ ۗ إِنَّ فِى ذَٰلِكَ لَـَٔايَةً لِّقَوْمٍ يَعْلَمُونَ ﴿٥٢﴾

    آخرها يشبه آية هذه السورة: «لِّقَوْمٍ يَعْلَمُونَ» بدل «لِّلْمُؤْمِنِينَ»

    العنكبوت ٤٤

    خَلَقَ ٱللَّهُ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضَ بِٱلْحَقِّ ۚ إِنَّ فِى ذَٰلِكَ لَـَٔايَةً لِّلْمُؤْمِنِينَ ﴿٤٤﴾

    زيادة «خَلَقَ ٱللَّهُ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضَ بِٱلْحَقِّ» في أولها

    سبإ ٩

    أَفَلَمْ يَرَوْا۟ إِلَىٰ مَا بَيْنَ أَيْدِيهِمْ وَمَا خَلْفَهُم مِّنَ ٱلسَّمَآءِ وَٱلْأَرْضِ ۚ إِن نَّشَأْ نَخْسِفْ بِهِمُ ٱلْأَرْضَ أَوْ نُسْقِطْ عَلَيْهِمْ كِسَفًا مِّنَ ٱلسَّمَآءِ ۚ إِنَّ فِى ذَٰلِكَ لَـَٔايَةً لِّكُلِّ عَبْدٍ مُّنِيبٍ ﴿٩﴾

    آخرها يشبه آية هذه السورة: «لِّكُلِّ عَبْدٍ مُّنِيبٍ» بدل «لِّلْمُؤْمِنِينَ»

  22. وَلَقَدْ كَذَّبَ أَصْحَـٰبُ ٱلْحِجْرِ ٱلْمُرْسَلِينَ ﴿٨٠﴾

    الشعراء ١٧٦

    كَذَّبَ أَصْحَـٰبُ لْـَٔيْكَةِ ٱلْمُرْسَلِينَ ﴿١٧٦﴾

    الشعراء ٢٠٧

    الزمر ٥٠

    قَدْ قَالَهَا ٱلَّذِينَ مِن قَبْلِهِمْ فَمَآ أَغْنَىٰ عَنْهُم مَّا كَانُوا۟ يَكْسِبُونَ ﴿٥٠﴾

  24. وَمَا خَلَقْنَا ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضَ وَمَا بَيْنَهُمَآ إِلَّا بِٱلْحَقِّ ۗ وَإِنَّ ٱلسَّاعَةَ لَـَٔاتِيَةٌ ۖ فَٱصْفَحِ ٱلصَّفْحَ ٱلْجَمِيلَ ﴿٨٥﴾

    الأنبياء ١٦
