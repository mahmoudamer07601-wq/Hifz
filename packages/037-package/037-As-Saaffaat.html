    رَبُّ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ وَمَا بَيْنَهُمَا ٱلْعَزِيزُ ٱلْغَفَّـٰرُ ﴿٦٦﴾

    «ٱلْعَزِيزُ ٱلْغَفَّـٰرُ» بدل «وَرَبُّ ٱلْمَشَـٰرِقِ»

    الدخان ٧

    رَبِّ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ وَمَا بَيْنَهُمَآ ۖ إِن كُنتُم مُّوقِنِينَ ﴿٧﴾

    «إِن كُنتُم مُّوقِنِينَ» بدل «وَرَبُّ ٱلْمَشَـٰرِقِ»

    النبإ ٣٧

    رَّبِّ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضِ وَمَا بَيْنَهُمَا ٱلرَّحْمَـٰنِ ۖ لَا يَمْلِكُونَ مِنْهُ خِطَابًا ﴿٣٧﴾

    «ٱلرَّحْمَـٰنِ لَا يَمْلِكُونَ مِنْهُ خِطَابًا» بدل «وَرَبُّ ٱلْمَشَـٰرِقِ»

  2. وَحِفْظًا مِّن كُلِّ شَيْطَـٰنٍ مَّارِدٍ ﴿٧﴾

    الحجر ١٧

    وَحَفِظْنَـٰهَا مِن كُلِّ شَيْطَـٰنٍ رَّجِيمٍ ﴿١٧﴾

    «وَحَفِظْنَـٰهَا» بدل «وَحِفْظًا»؛ «رَّجِيمٍ» بدل «مَّارِدٍ»

  3. وَقَالُوٓا۟ إِنْ هَـٰذَآ إِلَّا سِحْرٌ مُّبِينٌ ﴿١٥﴾

    المائدة ١١٠

    إِذْ قَالَ ٱللَّهُ يَـٰعِيسَى ٱبْنَ مَرْيَمَ ٱذْكُرْ نِعْمَتِى عَلَيْكَ وَعَلَىٰ وَٰلِدَتِكَ إِذْ أَيَّدتُّكَ بِرُوحِ ٱلْقُدُسِ تُكَلِّمُ ٱلنَّاسَ فِى ٱلْمَهْدِ وَكَهْلًا ۖ وَإِذْ عَلَّمْتُكَ ٱلْكِتَـٰبَ وَٱلْحِكْمَةَ وَٱلتَّوْرَىٰةَ وَٱلْإِنجِيلَ ۖ وَإِذْ تَخْلُقُ مِنَ ٱلطِّينِ كَهَيْـَٔةِ ٱلطَّيْرِ بِإِذْنِى فَتَنفُخُ فِيهَا فَتَكُونُ طَيْرًۢا بِإِذْنِى ۖ وَتُبْرِئُ ٱلْأَكْمَهَ وَٱلْأَبْرَصَ بِإِذْنِى ۖ وَإِذْ تُخْرِجُ ٱلْمَوْتَىٰ بِإِذْنِى ۖ وَإِذْ كَفَفْتُ بَنِىٓ إِسْرَٰٓءِيلَ عَنكَ إِذْ جِئْتَهُم بِٱلْبَيِّنَـٰتِ فَقَالَ ٱلَّذِينَ كَفَرُوا۟ مِنْهُمْ إِنْ هَـٰذَآ إِلَّا سِحْرٌ مُّبِينٌ ﴿١١٠﴾

    «إِذْ قَالَ ٱللَّهُ … كَفَرُوا۟ مِنْهُمْ» بدل «وَقَالُوٓا۟»

    الأنعام ٧

    وَلَوْ نَزَّلْنَا عَلَيْكَ كِتَـٰبًا فِى قِرْطَاسٍ فَلَمَسُوهُ بِأَيْدِيهِمْ لَقَالَ ٱلَّذِينَ كَفَرُوٓا۟ إِنْ هَـٰذَآ إِلَّا سِحْرٌ مُّبِينٌ ﴿٧﴾

    «وَلَوْ نَزَّلْنَا عَلَيْكَ … ٱلَّذِينَ كَفَرُوٓا۟» بدل «وَقَالُوٓا۟»

    يونس ٧٦

    فَلَمَّا جَآءَهُمُ ٱلْحَقُّ مِنْ عِندِنَا قَالُوٓا۟ إِنَّ هَـٰذَا لَسِحْرٌ مُّبِينٌ ﴿٧٦﴾

    آخرها يشبه آية هذه السورة: «قَالُوٓا۟» بدل «وَقَالُوٓا۟»؛ «لَسِحْرٌ» بدل «إِلَّا سِحْرٌ»

    هود ٧

    وَهُوَ ٱلَّذِى خَلَقَ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضَ فِى سِتَّةِ أَيَّامٍ وَكَانَ عَرْشُهُۥ عَلَى ٱلْمَآءِ لِيَبْلُوَكُمْ أَيُّكُمْ أَحْسَنُ عَمَلًا ۗ وَلَئِن قُلْتَ إِنَّكُم مَّبْعُوثُونَ مِنۢ بَعْدِ ٱلْمَوْتِ لَيَقُولَنَّ ٱلَّذِينَ كَفَرُوٓا۟ إِنْ هَـٰذَآ إِلَّا سِحْرٌ مُّبِينٌ ﴿٧﴾

    «وَهُوَ ٱلَّذِى خَلَقَ … ٱلَّذِينَ كَفَرُوٓا۟» بدل «وَقَالُوٓا۟»

    القصص ٣٦

    فَلَمَّا جَآءَهُم مُّوسَىٰ بِـَٔايَـٰتِنَا بَيِّنَـٰتٍ قَالُوا۟ مَا هَـٰذَآ إِلَّا سِحْرٌ مُّفْتَرًى وَمَا سَمِعْنَا بِهَـٰذَا فِىٓ ءَابَآئِنَا ٱلْأَوَّلِينَ ﴿٣٦﴾

    «فَلَمَّا جَآءَهُم مُّوسَىٰ بِـَٔايَـٰتِنَا بَيِّنَـٰتٍ قَالُوا۟ مَا» بدل «وَقَالُوٓا۟ إِنْ»؛ «مُّفْتَرًى وَمَا سَمِعْنَا بِهَـٰذَا فِىٓ ءَابَآئِنَا ٱلْأَوَّلِينَ» بدل «مُّبِينٌ»

    سبإ ٤٣

    وَإِذَا تُتْلَىٰ عَلَيْهِمْ ءَايَـٰتُنَا بَيِّنَـٰتٍ قَالُوا۟ مَا هَـٰذَآ إِلَّا رَجُلٌ يُرِيدُ أَن يَصُدَّكُمْ عَمَّا كَانَ يَعْبُدُ ءَابَآؤُكُمْ وَقَالُوا۟ مَا هَـٰذَآ إِلَّآ إِفْكٌ مُّفْتَرًى ۚ وَقَالَ ٱلَّذِينَ كَفَرُوا۟ لِلْحَقِّ لَمَّا جَآءَهُمْ إِنْ هَـٰذَآ إِلَّا سِحْرٌ مُّبِينٌ ﴿٤٣﴾

    آخرها يشبه آية هذه السورة: «قَالُوا۟ مَا هَـٰذَآ … لَمَّا جَآءَهُمْ» بدل «وَقَالُوٓا۟»

    المدثر ٢٤

    فَقَالَ إِنْ هَـٰذَآ إِلَّا سِحْرٌ يُؤْثَرُ ﴿٢٤﴾

    «فَقَالَ» بدل «وَقَالُوٓا۟»؛ «يُؤْثَرُ» بدل «مُّبِينٌ»

  4. أَءِذَا مِتْنَا وَكُنَّا تُرَابًا وَعِظَـٰمًا أَءِنَّا لَمَبْعُوثُونَ ﴿١٦﴾

    الإسراء ٤٩

    وَقَالُوٓا۟ أَءِذَا كُنَّا عِظَـٰمًا وَرُفَـٰتًا أَءِنَّا لَمَبْعُوثُونَ خَلْقًا جَدِيدًا ﴿٤٩﴾

    «وَقَالُوٓا۟ أَءِذَا كُنَّا عِظَـٰمًا وَرُفَـٰتًا» بدل «أَءِذَا مِتْنَا وَكُنَّا تُرَابًا وَعِظَـٰمًا»؛ زيادة «خَلْقًا جَدِيدًا» في آخرها

    الإسراء ٩٨

    ذَٰلِكَ جَزَآؤُهُم بِأَنَّهُمْ كَفَرُوا۟ بِـَٔايَـٰتِنَا وَقَالُوٓا۟ أَءِذَا كُنَّا عِظَـٰمًا وَرُفَـٰتًا أَءِنَّا لَمَبْعُوثُونَ خَلْقًا جَدِيدًا ﴿٩٨﴾

    في وسطها ما يشبه آية هذه السورة: «كُنَّا عِظَـٰمًا وَرُفَـٰتًا» بدل «مِتْنَا وَكُنَّا تُرَابًا وَعِظَـٰمًا»

    المؤمنون ٨٢

    قَالُوٓا۟ أَءِذَا مِتْنَا وَكُنَّا تُرَابًا وَعِظَـٰمًا أَءِنَّا لَمَبْعُوثُونَ ﴿٨٢﴾

    زيادة «قَالُوٓا۟» في أولها

    الواقعة ٤٧

    وَكَانُوا۟ يَقُولُونَ أَئِذَا مِتْنَا وَكُنَّا تُرَابًا وَعِظَـٰمًا أَءِنَّا لَمَبْعُوثُونَ ﴿٤٧﴾

    «وَكَانُوا۟ يَقُولُونَ أَئِذَا» بدل «أَءِذَا»

  5. فَإِنَّمَا هِىَ زَجْرَةٌ وَٰحِدَةٌ فَإِذَا هُمْ يَنظُرُونَ ﴿١٩﴾

    النازعات ١٣

    فَإِنَّمَا هِىَ زَجْرَةٌ وَٰحِدَةٌ ﴿١٣﴾

    حذف «فَإِذَا هُمْ يَنظُرُونَ» من آخرها

  6. وَقَالُوا۟ يَـٰوَيْلَنَا هَـٰذَا يَوْمُ ٱلدِّينِ ﴿٢٠﴾

    الواقعة ٥٦

    هَـٰذَا نُزُلُهُمْ يَوْمَ ٱلدِّينِ ﴿٥٦﴾

    «هَـٰذَا نُزُلُهُمْ» بدل «وَقَالُوا۟ يَـٰوَيْلَنَا هَـٰذَا»

  7. هَـٰذَا يَوْمُ ٱلْفَصْلِ ٱلَّذِى كُنتُم بِهِۦ تُكَذِّبُونَ ﴿٢١﴾

    المطففين ١٧

    ثُمَّ يُقَالُ هَـٰذَا ٱلَّذِى كُنتُم بِهِۦ تُكَذِّبُونَ ﴿١٧﴾

    «ثُمَّ يُقَالُ هَـٰذَا» بدل «هَـٰذَا يَوْمُ ٱلْفَصْلِ»

  8. وَأَقْبَلَ بَعْضُهُمْ عَلَىٰ بَعْضٍ يَتَسَآءَلُونَ ﴿٢٧﴾

    القلم ٣٠

    فَأَقْبَلَ بَعْضُهُمْ عَلَىٰ بَعْضٍ يَتَلَـٰوَمُونَ ﴿٣٠﴾

    «فَأَقْبَلَ» بدل «وَأَقْبَلَ»؛ «يَتَلَـٰوَمُونَ» بدل «يَتَسَآءَلُونَ»

  9. إِنَّا كَذَٰلِكَ نَفْعَلُ بِٱلْمُجْرِمِينَ ﴿٣٤﴾

    المرسلات ١٨

    كَذَٰلِكَ نَفْعَلُ بِٱلْمُجْرِمِينَ ﴿١٨﴾

    حذف «إِنَّا» من أولها

    المرسلات ٤٤

    إِنَّا كَذَٰلِكَ نَجْزِى ٱلْمُحْسِنِينَ ﴿٤٤﴾

    «نَجْزِى ٱلْمُحْسِنِينَ» بدل «نَفْعَلُ بِٱلْمُجْرِمِينَ»

  10. إِنَّهُمْ كَانُوٓا۟ إِذَا قِيلَ لَهُمْ لَآ إِلَـٰهَ إِلَّا ٱللَّهُ يَسْتَكْبِرُونَ ﴿٣٥﴾

    المرسلات ٤٨

    وَإِذَا قِيلَ لَهُمُ ٱرْكَعُوا۟ لَا يَرْكَعُونَ ﴿٤٨﴾

    «وَإِذَا» بدل «إِنَّهُمْ كَانُوٓا۟ إِذَا»؛ «ٱرْكَعُوا۟ لَا يَرْكَعُونَ» بدل «لَآ إِلَـٰهَ إِلَّا ٱللَّهُ يَسْتَكْبِرُونَ»

  11. وَمَا تُجْزَوْنَ إِلَّا مَا كُنتُمْ تَعْمَلُونَ ﴿٣٩﴾

    النمل ٩٠

    وَمَن جَآءَ بِٱلسَّيِّئَةِ فَكُبَّتْ وُجُوهُهُمْ فِى ٱلنَّارِ هَلْ تُجْزَوْنَ إِلَّا مَا كُنتُمْ تَعْمَلُونَ ﴿٩٠﴾

    «وَمَن جَآءَ بِٱلسَّيِّئَةِ فَكُبَّتْ وُجُوهُهُمْ فِى ٱلنَّارِ هَلْ» بدل «وَمَا»

    يس ٥٤

    فَٱلْيَوْمَ لَا تُظْلَمُ نَفْسٌ شَيْـًٔا وَلَا تُجْزَوْنَ إِلَّا مَا كُنتُمْ تَعْمَلُونَ ﴿٥٤﴾

    «فَٱلْيَوْمَ لَا تُظْلَمُ نَفْسٌ شَيْـًٔا وَلَا» بدل «وَمَا»

  12. إِلَّا عِبَادَ ٱللَّهِ ٱلْمُخْلَصِينَ ﴿٤٠﴾

    (وترد بلفظها أيضاً في ٧٤، ١٢٨، ١٦٠)

    الحجر ٤٠؛ ص ٨٣

    إِلَّا عِبَادَكَ مِنْهُمُ ٱلْمُخْلَصِينَ ﴿٤٠﴾

    «عِبَادَكَ مِنْهُمُ» بدل «عِبَادَ ٱللَّهِ»

  13. فِى جَنَّـٰتِ ٱلنَّعِيمِ ﴿٤٣﴾

    الشعراء ١٤٧؛ الدخان ٥٢

    فِى جَنَّـٰتٍ وَعُيُونٍ ﴿١٤٧﴾

    «وَعُيُونٍ» بدل «ٱلنَّعِيمِ»

    المدثر ٤٠

    فِى جَنَّـٰتٍ يَتَسَآءَلُونَ ﴿٤٠﴾

    «يَتَسَآءَلُونَ» بدل «ٱلنَّعِيمِ»

  14. عَلَىٰ سُرُرٍ مُّتَقَـٰبِلِينَ ﴿٤٤﴾

    الواقعة ١٥

    عَلَىٰ سُرُرٍ مَّوْضُونَةٍ ﴿١٥﴾

    «مَّوْضُونَةٍ» بدل «مُّتَقَـٰبِلِينَ»

  15. وَعِندَهُمْ قَـٰصِرَٰتُ ٱلطَّرْفِ عِينٌ ﴿٤٨﴾

    ص ٥٢

    وَعِندَهُمْ قَـٰصِرَٰتُ ٱلطَّرْفِ أَتْرَابٌ ﴿٥٢﴾

    «أَتْرَابٌ» بدل «عِينٌ»

  16. فَأَقْبَلَ بَعْضُهُمْ عَلَىٰ بَعْضٍ يَتَسَآءَلُونَ ﴿٥٠﴾

    الطور ٢٥

    وَأَقْبَلَ بَعْضُهُمْ عَلَىٰ بَعْضٍ يَتَسَآءَلُونَ ﴿٢٥﴾

    «وَأَقْبَلَ» بدل «فَأَقْبَلَ»

    القلم ٣٠

    فَأَقْبَلَ بَعْضُهُمْ عَلَىٰ بَعْضٍ يَتَلَـٰوَمُونَ ﴿٣٠﴾

    «يَتَلَـٰوَمُونَ» بدل «يَتَسَآءَلُونَ»

  17. أَءِذَا مِتْنَا وَكُنَّا تُرَابًا وَعِظَـٰمًا أَءِنَّا لَمَدِينُونَ ﴿٥٣﴾

    المؤمنون ٨٢

    قَالُوٓا۟ أَءِذَا مِتْنَا وَكُنَّا تُرَابًا وَعِظَـٰمًا أَءِنَّا لَمَبْعُوثُونَ ﴿٨٢﴾

    زيادة «قَالُوٓا۟» في أولها؛ «لَمَبْعُوثُونَ» بدل «لَمَدِينُونَ»

  18. إِلَّا مَوْتَتَنَا ٱلْأُولَىٰ وَمَا نَحْنُ بِمُعَذَّبِينَ ﴿٥٩﴾

    الدخان ٣٥

    إِنْ هِىَ إِلَّا مَوْتَتُنَا ٱلْأُولَىٰ وَمَا نَحْنُ بِمُنشَرِينَ ﴿٣٥﴾

    «إِنْ هِىَ إِلَّا مَوْتَتُنَا» بدل «إِلَّا مَوْتَتَنَا»؛ «بِمُنشَرِينَ» بدل «بِمُعَذَّبِينَ»

  19. إِنَّ هَـٰذَا لَهُوَ ٱلْفَوْزُ ٱلْعَظِيمُ ﴿٦٠﴾

    الواقعة ٩٥

    إِنَّ هَـٰذَا لَهُوَ حَقُّ ٱلْيَقِينِ ﴿٩٥﴾

    «حَقُّ ٱلْيَقِينِ» بدل «ٱلْفَوْزُ ٱلْعَظِيمُ»

  20. فَٱنظُرْ كَيْفَ كَانَ عَـٰقِبَةُ ٱلْمُنذَرِينَ ﴿٧٣﴾

    الأعراف ٨٤

    وَأَمْطَرْنَا عَلَيْهِم مَّطَرًا ۖ فَٱنظُرْ كَيْفَ كَانَ عَـٰقِبَةُ ٱلْمُجْرِمِينَ ﴿٨٤﴾

    آخرها يشبه آية هذه السورة: «ٱلْمُجْرِمِينَ» بدل «ٱلْمُنذَرِينَ»

    الأعراف ١٠٣

    ثُمَّ بَعَثْنَا مِنۢ بَعْدِهِم مُّوسَىٰ بِـَٔايَـٰتِنَآ إِلَىٰ فِرْعَوْنَ وَمَلَإِي۟هِۦ فَظَلَمُوا۟ بِهَا ۖ فَٱنظُرْ كَيْفَ كَانَ عَـٰقِبَةُ ٱلْمُفْسِدِينَ ﴿١٠٣﴾

    آخرها يشبه آية هذه السورة: «ٱلْمُفْسِدِينَ» بدل «ٱلْمُنذَرِينَ»

    يونس ٣٩

    بَلْ كَذَّبُوا۟ بِمَا لَمْ يُحِيطُوا۟ بِعِلْمِهِۦ وَلَمَّا يَأْتِهِمْ تَأْوِيلُهُۥ ۚ كَذَٰلِكَ كَذَّبَ ٱلَّذِينَ مِن قَبْلِهِمْ ۖ فَٱنظُرْ كَيْفَ كَانَ عَـٰقِبَةُ ٱلظَّـٰلِمِينَ ﴿٣٩﴾

    آخرها يشبه آية هذه السورة: «ٱلظَّـٰلِمِينَ» بدل «ٱلْمُنذَرِينَ»

    يونس ٧٣

    فَكَذَّبُوهُ فَنَجَّيْنَـٰهُ وَمَن مَّعَهُۥ فِى ٱلْفُلْكِ وَجَعَلْنَـٰهُمْ خَلَـٰٓئِفَ وَأَغْرَقْنَا ٱلَّذِينَ كَذَّبُوا۟ بِـَٔايَـٰتِنَا ۖ فَٱنظُرْ كَيْفَ كَانَ عَـٰقِبَةُ ٱلْمُنذَرِينَ ﴿٧٣﴾

    زيادة «فَكَذَّبُوهُ فَنَجَّيْنَـٰهُ وَمَن … كَذَّبُوا۟ بِـَٔايَـٰتِنَا» في أولها

    النمل ١٤

    وَجَحَدُوا۟ بِهَا وَٱسْتَيْقَنَتْهَآ أَنفُسُهُمْ ظُلْمًا وَعُلُوًّا ۚ فَٱنظُرْ كَيْفَ كَانَ عَـٰقِبَةُ ٱلْمُفْسِدِينَ ﴿١٤﴾

    آخرها يشبه آية هذه السورة: «ٱلْمُفْسِدِينَ» بدل «ٱلْمُنذَرِينَ»

    النمل ٥١

    فَٱنظُرْ كَيْفَ كَانَ عَـٰقِبَةُ مَكْرِهِمْ أَنَّا دَمَّرْنَـٰهُمْ وَقَوْمَهُمْ أَجْمَعِينَ ﴿٥١﴾

    «مَكْرِهِمْ أَنَّا دَمَّرْنَـٰهُمْ وَقَوْمَهُمْ أَجْمَعِينَ» بدل «ٱلْمُنذَرِينَ»

    القصص ٤٠

    فَأَخَذْنَـٰهُ وَجُنُودَهُۥ فَنَبَذْنَـٰهُمْ فِى ٱلْيَمِّ ۖ فَٱنظُرْ كَيْفَ كَانَ عَـٰقِبَةُ ٱلظَّـٰلِمِينَ ﴿٤٠﴾

    آخرها يشبه آية هذه السورة: «ٱلظَّـٰلِمِينَ» بدل «ٱلْمُنذَرِينَ»

    الزخرف ٢٥

    فَٱنتَقَمْنَا مِنْهُمْ ۖ فَٱنظُرْ كَيْفَ كَانَ عَـٰقِبَةُ ٱلْمُكَذِّبِينَ ﴿٢٥﴾

    زيادة «فَٱنتَقَمْنَا مِنْهُمْ» في أولها؛ «ٱلْمُكَذِّبِينَ» بدل «ٱلْمُنذَرِينَ»

  21. وَنَجَّيْنَـٰهُ وَأَهْلَهُۥ مِنَ ٱلْكَرْبِ ٱلْعَظِيمِ ﴿٧٦﴾

    الأنبياء ٧٦

    وَنُوحًا إِذْ نَادَىٰ مِن قَبْلُ فَٱسْتَجَبْنَا لَهُۥ فَنَجَّيْنَـٰهُ وَأَهْلَهُۥ مِنَ ٱلْكَرْبِ ٱلْعَظِيمِ ﴿٧٦﴾

    آخرها يشبه آية هذه السورة: «فَنَجَّيْنَـٰهُ» بدل «وَنَجَّيْنَـٰهُ»

  22. ثُمَّ أَغْرَقْنَا ٱلْـَٔاخَرِينَ ﴿٨٢﴾

    الشعراء ١٢٠

    ثُمَّ أَغْرَقْنَا بَعْدُ ٱلْبَاقِينَ ﴿١٢٠﴾

    «بَعْدُ ٱلْبَاقِينَ» بدل «ٱلْـَٔاخَرِينَ»

    الشعراء ١٧٢

    ثُمَّ دَمَّرْنَا ٱلْـَٔاخَرِينَ ﴿١٧٢﴾

    «دَمَّرْنَا» بدل «أَغْرَقْنَا»

  23. إِذْ قَالَ لِأَبِيهِ وَقَوْمِهِۦ مَاذَا تَعْبُدُونَ ﴿٨٥﴾

    الأنبياء ٥٢

    إِذْ قَالَ لِأَبِيهِ وَقَوْمِهِۦ مَا هَـٰذِهِ ٱلتَّمَاثِيلُ ٱلَّتِىٓ أَنتُمْ لَهَا عَـٰكِفُونَ ﴿٥٢﴾

    «مَا هَـٰذِهِ ٱلتَّمَاثِيلُ ٱلَّتِىٓ أَنتُمْ لَهَا عَـٰكِفُونَ» بدل «مَاذَا تَعْبُدُونَ»

    الشعراء ٧٠

    إِذْ قَالَ لِأَبِيهِ وَقَوْمِهِۦ مَا تَعْبُدُونَ ﴿٧٠﴾

    «مَا» بدل «مَاذَا»

  24. فَأَرَادُوا۟ بِهِۦ كَيْدًا فَجَعَلْنَـٰهُمُ ٱلْأَسْفَلِينَ ﴿٩٨﴾

    الأنبياء ٧٠

    وَأَرَادُوا۟ بِهِۦ كَيْدًا فَجَعَلْنَـٰهُمُ ٱلْأَخْسَرِينَ ﴿٧٠﴾

    «وَأَرَادُوا۟» بدل «فَأَرَادُوا۟»؛ «ٱلْأَخْسَرِينَ» بدل «ٱلْأَسْفَلِينَ»

  25. وَقَالَ إِنِّى ذَاهِبٌ إِلَىٰ رَبِّى سَيَهْدِينِ ﴿٩٩﴾

    العنكبوت ٢٦

    فَـَٔامَنَ لَهُۥ لُوطٌ ۘ وَقَالَ إِنِّى مُهَاجِرٌ إِلَىٰ رَبِّىٓ ۖ إِنَّهُۥ هُوَ ٱلْعَزِيزُ ٱلْحَكِيمُ ﴿٢٦﴾

    زيادة «فَـَٔامَنَ لَهُۥ لُوطٌ» في أولها؛ «مُهَاجِرٌ» بدل «ذَاهِبٌ»؛ «إِنَّهُۥ هُوَ ٱلْعَزِيزُ ٱلْحَكِيمُ» بدل «سَيَهْدِينِ»

  26. رَبِّ هَبْ لِى مِنَ ٱلصَّـٰلِحِينَ ﴿١٠٠﴾

    آل عمران ٣٨

    هُنَالِكَ دَعَا زَكَرِيَّا رَبَّهُۥ ۖ قَالَ رَبِّ هَبْ لِى مِن لَّدُنكَ ذُرِّيَّةً طَيِّبَةً ۖ إِنَّكَ سَمِيعُ ٱلدُّعَآءِ ﴿٣٨﴾

    زيادة «هُنَالِكَ دَعَا زَكَرِيَّا رَبَّهُۥ قَالَ» في أولها؛ «لَّدُنكَ ذُرِّيَّةً طَيِّبَةً إِنَّكَ سَمِيعُ ٱلدُّعَآءِ» بدل «ٱلصَّـٰلِحِينَ»

  27. قَدْ صَدَّقْتَ ٱلرُّءْيَآ ۚ إِنَّا كَذَٰلِكَ نَجْزِى ٱلْمُحْسِنِينَ ﴿١٠٥﴾

    المرسلات ٤٤

    إِنَّا كَذَٰلِكَ نَجْزِى ٱلْمُحْسِنِينَ ﴿٤٤﴾

    حذف «قَدْ صَدَّقْتَ ٱلرُّءْيَآ» من أولها

  28. إِنَّ هَـٰذَا لَهُوَ ٱلْبَلَـٰٓؤُا۟ ٱلْمُبِينُ ﴿١٠٦﴾

    النمل ١٦

    وَوَرِثَ سُلَيْمَـٰنُ دَاوُۥدَ ۖ وَقَالَ يَـٰٓأَيُّهَا ٱلنَّاسُ عُلِّمْنَا مَنطِقَ ٱلطَّيْرِ وَأُوتِينَا مِن كُلِّ شَىْءٍ ۖ إِنَّ هَـٰذَا لَهُوَ ٱلْفَضْلُ ٱلْمُبِينُ ﴿١٦﴾

    آخرها يشبه آية هذه السورة: «ٱلْفَضْلُ» بدل «ٱلْبَلَـٰٓؤُا۟»

    الواقعة ٩٥

    إِنَّ هَـٰذَا لَهُوَ حَقُّ ٱلْيَقِينِ ﴿٩٥﴾

    «حَقُّ ٱلْيَقِينِ» بدل «ٱلْبَلَـٰٓؤُا۟ ٱلْمُبِينُ»

  29. كَذَٰلِكَ نَجْزِى ٱلْمُحْسِنِينَ ﴿١١٠﴾

    المرسلات ٤٤

    إِنَّا كَذَٰلِكَ نَجْزِى ٱلْمُحْسِنِينَ ﴿٤٤﴾

    زيادة «إِنَّا» في أولها

  30. وَهَدَيْنَـٰهُمَا ٱلصِّرَٰطَ ٱلْمُسْتَقِيمَ ﴿١١٨﴾

    الفاتحة ٦

    ٱهْدِنَا ٱلصِّرَٰطَ ٱلْمُسْتَقِيمَ ﴿٦﴾

    «ٱهْدِنَا» بدل «وَهَدَيْنَـٰهُمَا»

  31. إِذْ قَالَ لِقَوْمِهِۦٓ أَلَا تَتَّقُونَ ﴿١٢٤﴾

    الشعراء ١٧٧

    إِذْ قَالَ لَهُمْ شُعَيْبٌ أَلَا تَتَّقُونَ ﴿١٧٧﴾

    «لَهُمْ شُعَيْبٌ» بدل «لِقَوْمِهِۦٓ»

  32. ٱللَّهَ رَبَّكُمْ وَرَبَّ ءَابَآئِكُمُ ٱلْأَوَّلِينَ ﴿١٢٦﴾

    الشعراء ٢٦

    قَالَ رَبُّكُمْ وَرَبُّ ءَابَآئِكُمُ ٱلْأَوَّلِينَ ﴿٢٦﴾

    «قَالَ رَبُّكُمْ» بدل «ٱللَّهَ رَبَّكُمْ»

  33. إِذْ نَجَّيْنَـٰهُ وَأَهْلَهُۥٓ أَجْمَعِينَ ﴿١٣٤﴾

    الشعراء ١٧٠

    فَنَجَّيْنَـٰهُ وَأَهْلَهُۥٓ أَجْمَعِينَ ﴿١٧٠﴾

    «فَنَجَّيْنَـٰهُ» بدل «إِذْ نَجَّيْنَـٰهُ»

  34. ثُمَّ دَمَّرْنَا ٱلْـَٔاخَرِينَ ﴿١٣٦﴾

    الشعراء ٦٦

    ثُمَّ أَغْرَقْنَا ٱلْـَٔاخَرِينَ ﴿٦٦﴾

    «أَغْرَقْنَا» بدل «دَمَّرْنَا»

  35. فَأْتُوا۟ بِكِتَـٰبِكُمْ إِن كُنتُمْ صَـٰدِقِينَ ﴿١٥٧﴾

    آل عمران ٩٣

    كُلُّ ٱلطَّعَامِ كَانَ حِلًّا لِّبَنِىٓ إِسْرَٰٓءِيلَ إِلَّا مَا حَرَّمَ إِسْرَٰٓءِيلُ عَلَىٰ نَفْسِهِۦ مِن قَبْلِ أَن تُنَزَّلَ ٱلتَّوْرَىٰةُ ۗ قُلْ فَأْتُوا۟ بِٱلتَّوْرَىٰةِ فَٱتْلُوهَآ إِن كُنتُمْ صَـٰدِقِينَ ﴿٩٣﴾

    آخرها يشبه آية هذه السورة: «بِٱلتَّوْرَىٰةِ فَٱتْلُوهَآ» بدل «بِكِتَـٰبِكُمْ»

    الدخان ٣٦

    فَأْتُوا۟ بِـَٔابَآئِنَآ إِن كُنتُمْ صَـٰدِقِينَ ﴿٣٦﴾

    «بِـَٔابَآئِنَآ» بدل «بِكِتَـٰبِكُمْ»

    الواقعة ٨٧

    تَرْجِعُونَهَآ إِن كُنتُمْ صَـٰدِقِينَ ﴿٨٧﴾

    «تَرْجِعُونَهَآ» بدل «فَأْتُوا۟ بِكِتَـٰبِكُمْ»

  36. سُبْحَـٰنَ ٱللَّهِ عَمَّا يَصِفُونَ ﴿١٥٩﴾

    المؤمنون ٩١

    مَا ٱتَّخَذَ ٱللَّهُ مِن وَلَدٍ وَمَا كَانَ مَعَهُۥ مِنْ إِلَـٰهٍ ۚ إِذًا لَّذَهَبَ كُلُّ إِلَـٰهٍۭ بِمَا خَلَقَ وَلَعَلَا بَعْضُهُمْ عَلَىٰ بَعْضٍ ۚ سُبْحَـٰنَ ٱللَّهِ عَمَّا يَصِفُونَ ﴿٩١﴾

    زيادة «مَا ٱتَّخَذَ ٱللَّهُ … عَلَىٰ بَعْضٍ» في أولها

  37. سُبْحَـٰنَ رَبِّكَ رَبِّ ٱلْعِزَّةِ عَمَّا يَصِفُونَ ﴿١٨٠﴾

    الأنبياء ٢٢

    لَوْ كَانَ فِيهِمَآ ءَالِهَةٌ إِلَّا ٱللَّهُ لَفَسَدَتَا ۚ فَسُبْحَـٰنَ ٱللَّهِ رَبِّ ٱلْعَرْشِ عَمَّا يَصِفُونَ ﴿٢٢﴾

    آخرها يشبه آية هذه السورة: «فَسُبْحَـٰنَ ٱللَّهِ» بدل «سُبْحَـٰنَ رَبِّكَ»؛ «ٱلْعَرْشِ» بدل «ٱلْعِزَّةِ»

  38. وَٱلْحَمْدُ لِلَّهِ رَبِّ ٱلْعَـٰلَمِينَ ﴿١٨٢﴾

    الفاتحة ٢

    ٱلْحَمْدُ لِلَّهِ رَبِّ ٱلْعَـٰلَمِينَ ﴿٢﴾

    «ٱلْحَمْدُ» بدل «وَٱلْحَمْدُ»

    الأنعام ٤٥

    فَقُطِعَ دَابِرُ ٱلْقَوْمِ ٱلَّذِينَ ظَلَمُوا۟ ۚ وَٱلْحَمْدُ لِلَّهِ رَبِّ ٱلْعَـٰلَمِينَ ﴿٤٥﴾

  21. وَنَجَّيْنَـٰهُ وَأَهْلَهُۥ مِنَ ٱلْكَرْبِ ٱلْعَظِيمِ ﴿٧٦﴾

    الأنبياء 76

    وَنُوحًا إِذْ نَادَىٰ مِن قَبْلُ فَٱسْتَجَبْنَا لَهُۥ فَنَجَّيْنَـٰهُ وَأَهْلَهُۥ مِنَ ٱلْكَرْبِ ٱلْعَظِيمِ ﴿٧٦﴾

    Su final se parece a la aleya de esta sura: «فَنَجَّيْنَـٰهُ» en lugar de «وَنَجَّيْنَـٰهُ»

  22. ثُمَّ أَغْرَقْنَا ٱلْـَٔاخَرِينَ ﴿٨٢﴾

    الشعراء 120

    ثُمَّ أَغْرَقْنَا بَعْدُ ٱلْبَاقِينَ ﴿١٢٠﴾

    «بَعْدُ ٱلْبَاقِينَ» en lugar de «ٱلْـَٔاخَرِينَ»

    الشعراء 172

    ثُمَّ دَمَّرْنَا ٱلْـَٔاخَرِينَ ﴿١٧٢﴾

    «دَمَّرْنَا» en lugar de «أَغْرَقْنَا»

  23. إِذْ قَالَ لِأَبِيهِ وَقَوْمِهِۦ مَاذَا تَعْبُدُونَ ﴿٨٥﴾

    الأنبياء 52

    إِذْ قَالَ لِأَبِيهِ وَقَوْمِهِۦ مَا هَـٰذِهِ ٱلتَّمَاثِيلُ ٱلَّتِىٓ أَنتُمْ لَهَا عَـٰكِفُونَ ﴿٥٢﴾

    «مَا هَـٰذِهِ ٱلتَّمَاثِيلُ ٱلَّتِىٓ أَنتُمْ لَهَا عَـٰكِفُونَ» en lugar de «مَاذَا تَعْبُدُونَ»

    الشعراء 70

    إِذْ قَالَ لِأَبِيهِ وَقَوْمِهِۦ مَا تَعْبُدُونَ ﴿٧٠﴾

    «مَا» en lugar de «مَاذَا»

  24. فَأَرَادُوا۟ بِهِۦ كَيْدًا فَجَعَلْنَـٰهُمُ ٱلْأَسْفَلِينَ ﴿٩٨﴾

    الأنبياء 70

    وَأَرَادُوا۟ بِهِۦ كَيْدًا فَجَعَلْنَـٰهُمُ ٱلْأَخْسَرِينَ ﴿٧٠﴾

    «وَأَرَادُوا۟» en lugar de «فَأَرَادُوا۟»; «ٱلْأَخْسَرِينَ» en lugar de «ٱلْأَسْفَلِينَ»

  25. وَقَالَ إِنِّى ذَاهِبٌ إِلَىٰ رَبِّى سَيَهْدِينِ ﴿٩٩﴾

    العنكبوت 26

    فَـَٔامَنَ لَهُۥ لُوطٌ ۘ وَقَالَ إِنِّى مُهَاجِرٌ إِلَىٰ رَبِّىٓ ۖ إِنَّهُۥ هُوَ ٱلْعَزِيزُ ٱلْحَكِيمُ ﴿٢٦﴾

    Se añade «فَـَٔامَنَ لَهُۥ لُوطٌ» al principio; «مُهَاجِرٌ» en lugar de «ذَاهِبٌ»; «إِنَّهُۥ هُوَ ٱلْعَزِيزُ ٱلْحَكِيمُ» en lugar de «سَيَهْدِينِ»

  26. رَبِّ هَبْ لِى مِنَ ٱلصَّـٰلِحِينَ ﴿١٠٠﴾

    آل عمران 38

    هُنَالِكَ دَعَا زَكَرِيَّا رَبَّهُۥ ۖ قَالَ رَبِّ هَبْ لِى مِن لَّدُنكَ ذُرِّيَّةً طَيِّبَةً ۖ إِنَّكَ سَمِيعُ ٱلدُّعَآءِ ﴿٣٨﴾

    Se añade «هُنَالِكَ دَعَا زَكَرِيَّا رَبَّهُۥ قَالَ» al principio; «لَّدُنكَ ذُرِّيَّةً طَيِّبَةً إِنَّكَ سَمِيعُ ٱلدُّعَآءِ» en lugar de «ٱلصَّـٰلِحِينَ»

  27. قَدْ صَدَّقْتَ ٱلرُّءْيَآ ۚ إِنَّا كَذَٰلِكَ نَجْزِى ٱلْمُحْسِنِينَ ﴿١٠٥﴾

    المرسلات 44

    إِنَّا كَذَٰلِكَ نَجْزِى ٱلْمُحْسِنِينَ ﴿٤٤﴾

    Se suprime «قَدْ صَدَّقْتَ ٱلرُّءْيَآ» del principio

  28. إِنَّ هَـٰذَا لَهُوَ ٱلْبَلَـٰٓؤُا۟ ٱلْمُبِينُ ﴿١٠٦﴾

    النمل 16

    وَوَرِثَ سُلَيْمَـٰنُ دَاوُۥدَ ۖ وَقَالَ يَـٰٓأَيُّهَا ٱلنَّاسُ عُلِّمْنَا مَنطِقَ ٱلطَّيْرِ وَأُوتِينَا مِن كُلِّ شَىْءٍ ۖ إِنَّ هَـٰذَا لَهُوَ ٱلْفَضْلُ ٱلْمُبِينُ ﴿١٦﴾

    Su final se parece a la aleya de esta sura: «ٱلْفَضْلُ» en lugar de «ٱلْبَلَـٰٓؤُا۟»

    الواقعة 95

    إِنَّ هَـٰذَا لَهُوَ حَقُّ ٱلْيَقِينِ ﴿٩٥﴾

    «حَقُّ ٱلْيَقِينِ» en lugar de «ٱلْبَلَـٰٓؤُا۟ ٱلْمُبِينُ»

  29. كَذَٰلِكَ نَجْزِى ٱلْمُحْسِنِينَ ﴿١١٠﴾

    المرسلات 44

    إِنَّا كَذَٰلِكَ نَجْزِى ٱلْمُحْسِنِينَ ﴿٤٤﴾

    Se añade «إِنَّا» al principio

  30. وَهَدَيْنَـٰهُمَا ٱلصِّرَٰطَ ٱلْمُسْتَقِيمَ ﴿١١٨﴾

    الفاتحة 6

    ٱهْدِنَا ٱلصِّرَٰطَ ٱلْمُسْتَقِيمَ ﴿٦﴾

    «ٱهْدِنَا» en lugar de «وَهَدَيْنَـٰهُمَا»

  31. إِذْ قَالَ لِقَوْمِهِۦٓ أَلَا تَتَّقُونَ ﴿١٢٤﴾

    الشعراء 177

    إِذْ قَالَ لَهُمْ شُعَيْبٌ أَلَا تَتَّقُونَ ﴿١٧٧﴾

    «لَهُمْ شُعَيْبٌ» en lugar de «لِقَوْمِهِۦٓ»

  32. ٱللَّهَ رَبَّكُمْ وَرَبَّ ءَابَآئِكُمُ ٱلْأَوَّلِينَ ﴿١٢٦﴾

    الشعراء 26

    قَالَ رَبُّكُمْ وَرَبُّ ءَابَآئِكُمُ ٱلْأَوَّلِينَ ﴿٢٦﴾

    «قَالَ رَبُّكُمْ» en lugar de «ٱللَّهَ رَبَّكُمْ»

  33. إِذْ نَجَّيْنَـٰهُ وَأَهْلَهُۥٓ أَجْمَعِينَ ﴿١٣٤﴾

    الشعراء 170

    فَنَجَّيْنَـٰهُ وَأَهْلَهُۥٓ أَجْمَعِينَ ﴿١٧٠﴾

    «فَنَجَّيْنَـٰهُ» en lugar de «إِذْ نَجَّيْنَـٰهُ»

  34. ثُمَّ دَمَّرْنَا ٱلْـَٔاخَرِينَ ﴿١٣٦﴾

    الشعراء 66

    ثُمَّ أَغْرَقْنَا ٱلْـَٔاخَرِينَ ﴿٦٦﴾

    «أَغْرَقْنَا» en lugar de «دَمَّرْنَا»

  35. فَأْتُوا۟ بِكِتَـٰبِكُمْ إِن كُنتُمْ صَـٰدِقِينَ ﴿١٥٧﴾

    آل عمران 93

    كُلُّ ٱلطَّعَامِ كَانَ حِلًّا لِّبَنِىٓ إِسْرَٰٓءِيلَ إِلَّا مَا حَرَّمَ إِسْرَٰٓءِيلُ عَلَىٰ نَفْسِهِۦ مِن قَبْلِ أَن تُنَزَّلَ ٱلتَّوْرَىٰةُ ۗ قُلْ فَأْتُوا۟ بِٱلتَّوْرَىٰةِ فَٱتْلُوهَآ إِن كُنتُمْ صَـٰدِقِينَ ﴿٩٣﴾

    Su final se parece a la aleya de esta sura: «بِٱلتَّوْرَىٰةِ فَٱتْلُوهَآ» en lugar de «بِكِتَـٰبِكُمْ»

    الدخان 36

    فَأْتُوا۟ بِـَٔابَآئِنَآ إِن كُنتُمْ صَـٰدِقِينَ ﴿٣٦﴾

    «بِـَٔابَآئِنَآ» en lugar de «بِكِتَـٰبِكُمْ»

    الواقعة 87

    تَرْجِعُونَهَآ إِن كُنتُمْ صَـٰدِقِينَ ﴿٨٧﴾

    «تَرْجِعُونَهَآ» en lugar de «فَأْتُوا۟ بِكِتَـٰبِكُمْ»

  36. سُبْحَـٰنَ ٱللَّهِ عَمَّا يَصِفُونَ ﴿١٥٩﴾

    المؤمنون 91

    مَا ٱتَّخَذَ ٱللَّهُ مِن وَلَدٍ وَمَا كَانَ مَعَهُۥ مِنْ إِلَـٰهٍ ۚ إِذًا لَّذَهَبَ كُلُّ إِلَـٰهٍۭ بِمَا خَلَقَ وَلَعَلَا بَعْضُهُمْ عَلَىٰ بَعْضٍ ۚ سُبْحَـٰنَ ٱللَّهِ عَمَّا يَصِفُونَ ﴿٩١﴾

    Se añade «مَا ٱتَّخَذَ ٱللَّهُ … عَلَىٰ بَعْضٍ» al principio

  37. سُبْحَـٰنَ رَبِّكَ رَبِّ ٱلْعِزَّةِ عَمَّا يَصِفُونَ ﴿١٨٠﴾

    الأنبياء 22

    لَوْ كَانَ فِيهِمَآ ءَالِهَةٌ إِلَّا ٱللَّهُ لَفَسَدَتَا ۚ فَسُبْحَـٰنَ ٱللَّهِ رَبِّ ٱلْعَرْشِ عَمَّا يَصِفُونَ ﴿٢٢﴾

    Su final se parece a la aleya de esta sura: «فَسُبْحَـٰنَ ٱللَّهِ» en lugar de «سُبْحَـٰنَ رَبِّكَ»; «ٱلْعَرْشِ» en lugar de «ٱلْعِزَّةِ»

  38. وَٱلْحَمْدُ لِلَّهِ رَبِّ ٱلْعَـٰلَمِينَ ﴿١٨٢﴾

    الفاتحة 2

    ٱلْحَمْدُ لِلَّهِ رَبِّ ٱلْعَـٰلَمِينَ ﴿٢﴾

    «ٱلْحَمْدُ» en lugar de «وَٱلْحَمْدُ»

    الأنعام 45

    فَقُطِعَ دَابِرُ ٱلْقَوْمِ ٱلَّذِينَ ظَلَمُوا۟ ۚ وَٱلْحَمْدُ لِلَّهِ رَبِّ ٱلْعَـٰلَمِينَ ﴿٤٥﴾

    Se añade «فَقُطِعَ دَابِرُ ٱلْقَوْمِ ٱلَّذِينَ ظَلَمُوا۟» al principio

    يونس 10

    دَعْوَىٰهُمْ فِيهَا سُبْحَـٰنَكَ ٱللَّهُمَّ وَتَحِيَّتُهُمْ فِيهَا سَلَـٰمٌ ۚ وَءَاخِرُ دَعْوَىٰهُمْ أَنِ ٱلْحَمْدُ لِلَّهِ رَبِّ ٱلْعَـٰلَمِينَ ﴿١٠﴾

    Su final se parece a la aleya de esta sura: «ٱلْحَمْدُ» en lugar de «وَٱلْحَمْدُ»

    الزمر 75

    وَتَرَى ٱلْمَلَـٰٓئِكَةَ حَآفِّينَ مِنْ حَوْلِ ٱلْعَرْشِ يُسَبِّحُونَ بِحَمْدِ رَبِّهِمْ ۖ وَقُضِىَ بَيْنَهُم بِٱلْحَقِّ وَقِيلَ ٱلْحَمْدُ لِلَّهِ رَبِّ ٱلْعَـٰلَمِينَ ﴿٧٥﴾

    Su final se parece a la aleya de esta sura: «ٱلْحَمْدُ» en lugar de «وَٱلْحَمْدُ»

    غافر 65

    هُوَ ٱلْحَىُّ لَآ إِلَـٰهَ إِلَّا هُوَ فَٱدْعُوهُ مُخْلِصِينَ لَهُ ٱلدِّينَ ۗ ٱلْحَمْدُ لِلَّهِ رَبِّ ٱلْعَـٰلَمِينَ ﴿٦٥﴾

    Su final se parece a la aleya de esta sura: «ٱلْحَمْدُ» en lugar de «وَٱلْحَمْدُ»
