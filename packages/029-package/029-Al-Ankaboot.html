    وَلَقَدْ فَتَنَّا ٱلَّذِينَ مِن قَبْلِهِمْ ۖ فَلَيَعْلَمَنَّ ٱللَّهُ ٱلَّذِينَ صَدَقُوا۟ وَلَيَعْلَمَنَّ ٱلْكَـٰذِبِينَ ﴿٣﴾

    وَلَيَعْلَمَنَّ ٱللَّهُ ٱلَّذِينَ ءَامَنُوا۟ وَلَيَعْلَمَنَّ ٱلْمُنَـٰفِقِينَ ﴿١١﴾

٣ · آيات تشترك في عبارة ثم تختلف

(٣)

  1. ١ آيتان تشتركان في «مِن دُونِ ٱللَّهِ أَوْثَـٰنًا»

    إِنَّمَا تَعْبُدُونَ مِن دُونِ ٱللَّهِ أَوْثَـٰنًا وَتَخْلُقُونَ إِفْكًا ۚ إِنَّ ٱلَّذِينَ … ﴿١٧﴾

    وَقَالَ إِنَّمَا ٱتَّخَذْتُم مِّن دُونِ ٱللَّهِ أَوْثَـٰنًا مَّوَدَّةَ بَيْنِكُمْ فِى ٱلْحَيَوٰةِ … ﴿٢٥﴾

    قبلها «ٱتَّخَذْتُم» بدل «تَعْبُدُونَ»؛ بعدها «مَّوَدَّةَ» بدل «وَتَخْلُقُونَ»

  2. ٢ آيتان تشتركان في «فَمَا كَانَ جَوَابَ قَوْمِهِۦٓ إِلَّآ أَن قَالُوا۟»

    فَمَا كَانَ جَوَابَ قَوْمِهِۦٓ إِلَّآ أَن قَالُوا۟ ٱقْتُلُوهُ أَوْ حَرِّقُوهُ فَأَنجَىٰهُ … ﴿٢٤﴾

    … وَتَأْتُونَ فِى نَادِيكُمُ ٱلْمُنكَرَ ۖ فَمَا كَانَ جَوَابَ قَوْمِهِۦٓ إِلَّآ أَن قَالُوا۟ ٱئْتِنَا بِعَذَابِ ٱللَّهِ إِن … ﴿٢٩﴾

    قبلها «ٱلْمُنكَرَ»؛ بعدها «ٱئْتِنَا» بدل «ٱقْتُلُوهُ»

  3. ٣ آيتان تشتركان في «وَمَا يَجْحَدُ بِـَٔايَـٰتِنَآ إِلَّا»

    … هَـٰٓؤُلَآءِ مَن يُؤْمِنُ بِهِۦ ۚ وَمَا يَجْحَدُ بِـَٔايَـٰتِنَآ إِلَّا ٱلْكَـٰفِرُونَ ﴿٤٧﴾

    بَلْ هُوَ ءَايَـٰتٌۢ بَيِّنَـٰتٌ فِى صُدُورِ ٱلَّذِينَ أُوتُوا۟ ٱلْعِلْمَ ۚ وَمَا يَجْحَدُ بِـَٔايَـٰتِنَآ إِلَّا ٱلظَّـٰلِمُونَ ﴿٤٩﴾

    قبلها «ٱلْعِلْمَ» بدل «بِهِۦ»؛ بعدها «ٱلظَّـٰلِمُونَ» بدل «ٱلْكَـٰفِرُونَ»

آيات تشبهها في سور أخرى

  1. وَٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّـٰلِحَـٰتِ لَنُدْخِلَنَّهُمْ فِى ٱلصَّـٰلِحِينَ ﴿٩﴾

    الحج ٥٦

    ٱلْمُلْكُ يَوْمَئِذٍ لِّلَّهِ يَحْكُمُ بَيْنَهُمْ ۚ فَٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّـٰلِحَـٰتِ فِى جَنَّـٰتِ ٱلنَّعِيمِ ﴿٥٦﴾

    آخرها يشبه آية هذه السورة: «فَٱلَّذِينَ» بدل «وَٱلَّذِينَ»؛ «فِى جَنَّـٰتِ ٱلنَّعِيمِ» بدل «لَنُدْخِلَنَّهُمْ فِى ٱلصَّـٰلِحِينَ»

    النور ٥٥

    وَعَدَ ٱللَّهُ ٱلَّذِينَ ءَامَنُوا۟ مِنكُمْ وَعَمِلُوا۟ ٱلصَّـٰلِحَـٰتِ لَيَسْتَخْلِفَنَّهُمْ فِى ٱلْأَرْضِ كَمَا ٱسْتَخْلَفَ ٱلَّذِينَ مِن قَبْلِهِمْ وَلَيُمَكِّنَنَّ لَهُمْ دِينَهُمُ ٱلَّذِى ٱرْتَضَىٰ لَهُمْ وَلَيُبَدِّلَنَّهُم مِّنۢ بَعْدِ خَوْفِهِمْ أَمْنًا ۚ يَعْبُدُونَنِى لَا يُشْرِكُونَ بِى شَيْـًٔا ۚ وَمَن كَفَرَ بَعْدَ ذَٰلِكَ فَأُو۟لَـٰٓئِكَ هُمُ ٱلْفَـٰسِقُونَ ﴿٥٥﴾

    «وَعَدَ ٱللَّهُ ٱلَّذِينَ ءَامَنُوا۟ مِنكُمْ» بدل «وَٱلَّذِينَ ءَامَنُوا۟»؛ «لَيَسْتَخْلِفَنَّهُمْ» بدل «لَنُدْخِلَنَّهُمْ»؛ «ٱلْأَرْضِ كَمَا ٱسْتَخْلَفَ … هُمُ ٱلْفَـٰسِقُونَ» بدل «ٱلصَّـٰلِحِينَ»

    الروم ١٥

    فَأَمَّا ٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّـٰلِحَـٰتِ فَهُمْ فِى رَوْضَةٍ يُحْبَرُونَ ﴿١٥﴾

    «فَأَمَّا ٱلَّذِينَ» بدل «وَٱلَّذِينَ»؛ «فَهُمْ» بدل «لَنُدْخِلَنَّهُمْ»؛ «رَوْضَةٍ يُحْبَرُونَ» بدل «ٱلصَّـٰلِحِينَ»

    ص ٢٨

    أَمْ نَجْعَلُ ٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّـٰلِحَـٰتِ كَٱلْمُفْسِدِينَ فِى ٱلْأَرْضِ أَمْ نَجْعَلُ ٱلْمُتَّقِينَ كَٱلْفُجَّارِ ﴿٢٨﴾

    «أَمْ نَجْعَلُ ٱلَّذِينَ» بدل «وَٱلَّذِينَ»؛ «كَٱلْمُفْسِدِينَ» بدل «لَنُدْخِلَنَّهُمْ»؛ «ٱلْأَرْضِ أَمْ نَجْعَلُ ٱلْمُتَّقِينَ كَٱلْفُجَّارِ» بدل «ٱلصَّـٰلِحِينَ»

    الشورى ٢٢

    تَرَى ٱلظَّـٰلِمِينَ مُشْفِقِينَ مِمَّا كَسَبُوا۟ وَهُوَ وَاقِعٌۢ بِهِمْ ۗ وَٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّـٰلِحَـٰتِ فِى رَوْضَاتِ ٱلْجَنَّاتِ ۖ لَهُم مَّا يَشَآءُونَ عِندَ رَبِّهِمْ ۚ ذَٰلِكَ هُوَ ٱلْفَضْلُ ٱلْكَبِيرُ ﴿٢٢﴾

    زيادة «تَرَى ٱلظَّـٰلِمِينَ مُشْفِقِينَ مِمَّا كَسَبُوا۟ وَهُوَ وَاقِعٌۢ بِهِمْ» في أولها؛ «فِى رَوْضَاتِ ٱلْجَنَّاتِ … ٱلْفَضْلُ ٱلْكَبِيرُ» بدل «لَنُدْخِلَنَّهُمْ فِى ٱلصَّـٰلِحِينَ»

    الجاثية ٣٠

    فَأَمَّا ٱلَّذِينَ ءَامَنُوا۟ وَعَمِلُوا۟ ٱلصَّـٰلِحَـٰتِ فَيُدْخِلُهُمْ رَبُّهُمْ فِى رَحْمَتِهِۦ ۚ ذَٰلِكَ هُوَ ٱلْفَوْزُ ٱلْمُبِينُ ﴿٣٠﴾

    «فَأَمَّا ٱلَّذِينَ» بدل «وَٱلَّذِينَ»؛ «فَيُدْخِلُهُمْ رَبُّهُمْ» بدل «لَنُدْخِلَنَّهُمْ»؛ «رَحْمَتِهِۦ ذَٰلِكَ هُوَ ٱلْفَوْزُ ٱلْمُبِينُ» بدل «ٱلصَّـٰلِحِينَ»

  2. وَإِن تُكَذِّبُوا۟ فَقَدْ كَذَّبَ أُمَمٌ مِّن قَبْلِكُمْ ۖ وَمَا عَلَى ٱلرَّسُولِ إِلَّا ٱلْبَلَـٰغُ ٱلْمُبِينُ ﴿١٨﴾

    يس ١٧

    وَمَا عَلَيْنَآ إِلَّا ٱلْبَلَـٰغُ ٱلْمُبِينُ ﴿١٧﴾

    تشبه آخرَ آية هذه السورة: «عَلَيْنَآ» بدل «عَلَى ٱلرَّسُولِ»

  3. قُلْ سِيرُوا۟ فِى ٱلْأَرْضِ فَٱنظُرُوا۟ كَيْفَ بَدَأَ ٱلْخَلْقَ ۚ ثُمَّ ٱللَّهُ يُنشِئُ ٱلنَّشْأَةَ ٱلْـَٔاخِرَةَ ۚ إِنَّ ٱللَّهَ عَلَىٰ كُلِّ شَىْءٍ قَدِيرٌ ﴿٢٠﴾

    النمل ٦٩

    قُلْ سِيرُوا۟ فِى ٱلْأَرْضِ فَٱنظُرُوا۟ كَيْفَ كَانَ عَـٰقِبَةُ ٱلْمُجْرِمِينَ ﴿٦٩﴾

    «كَانَ عَـٰقِبَةُ ٱلْمُجْرِمِينَ» بدل «بَدَأَ ٱلْخَلْقَ ثُمَّ … شَىْءٍ قَدِيرٌ»

  4. وَمَآ أَنتُم بِمُعْجِزِينَ فِى ٱلْأَرْضِ وَلَا فِى ٱلسَّمَآءِ ۖ وَمَا لَكُم مِّن دُونِ ٱللَّهِ مِن وَلِىٍّ وَلَا نَصِيرٍ ﴿٢٢﴾

    الشورى ٣١

    وَمَآ أَنتُم بِمُعْجِزِينَ فِى ٱلْأَرْضِ ۖ وَمَا لَكُم مِّن دُونِ ٱللَّهِ مِن وَلِىٍّ وَلَا نَصِيرٍ ﴿٣١﴾

    حذف «وَلَا فِى ٱلسَّمَآءِ»

  5. فَمَا كَانَ جَوَابَ قَوْمِهِۦٓ إِلَّآ أَن قَالُوا۟ ٱقْتُلُوهُ أَوْ حَرِّقُوهُ فَأَنجَىٰهُ ٱللَّهُ مِنَ ٱلنَّارِ ۚ إِنَّ فِى ذَٰلِكَ لَـَٔايَـٰتٍ لِّقَوْمٍ يُؤْمِنُونَ ﴿٢٤﴾

    الحجر ٧٥

    إِنَّ فِى ذَٰلِكَ لَـَٔايَـٰتٍ لِّلْمُتَوَسِّمِينَ ﴿٧٥﴾

    تشبه آخرَ آية هذه السورة: «لِّلْمُتَوَسِّمِينَ» بدل «لِّقَوْمٍ يُؤْمِنُونَ»

  6. فَـَٔامَنَ لَهُۥ لُوطٌ ۘ وَقَالَ إِنِّى مُهَاجِرٌ إِلَىٰ رَبِّىٓ ۖ إِنَّهُۥ هُوَ ٱلْعَزِيزُ ٱلْحَكِيمُ ﴿٢٦﴾

    الصافات ٩٩

    وَقَالَ إِنِّى ذَاهِبٌ إِلَىٰ رَبِّى سَيَهْدِينِ ﴿٩٩﴾

    حذف «فَـَٔامَنَ لَهُۥ لُوطٌ» من أولها؛ «ذَاهِبٌ» بدل «مُهَاجِرٌ»؛ «سَيَهْدِينِ» بدل «إِنَّهُۥ هُوَ ٱلْعَزِيزُ ٱلْحَكِيمُ»

  7. وَوَهَبْنَا لَهُۥٓ إِسْحَـٰقَ وَيَعْقُوبَ وَجَعَلْنَا فِى ذُرِّيَّتِهِ ٱلنُّبُوَّةَ وَٱلْكِتَـٰبَ وَءَاتَيْنَـٰهُ أَجْرَهُۥ فِى ٱلدُّنْيَا ۖ وَإِنَّهُۥ فِى ٱلْـَٔاخِرَةِ لَمِنَ ٱلصَّـٰلِحِينَ ﴿٢٧﴾

    النحل ١٢٢

    وَءَاتَيْنَـٰهُ فِى ٱلدُّنْيَا حَسَنَةً ۖ وَإِنَّهُۥ فِى ٱلْـَٔاخِرَةِ لَمِنَ ٱلصَّـٰلِحِينَ ﴿١٢٢﴾

    تشبه آخرَ آية هذه السورة: حذف «أَجْرَهُۥ»؛ زيادة «حَسَنَةً»

  8. وَلُوطًا إِذْ قَالَ لِقَوْمِهِۦٓ إِنَّكُمْ لَتَأْتُونَ ٱلْفَـٰحِشَةَ مَا سَبَقَكُم بِهَا مِنْ أَحَدٍ مِّنَ ٱلْعَـٰلَمِينَ ﴿٢٨﴾

    الأعراف ٨٠

    وَلُوطًا إِذْ قَالَ لِقَوْمِهِۦٓ أَتَأْتُونَ ٱلْفَـٰحِشَةَ مَا سَبَقَكُم بِهَا مِنْ أَحَدٍ مِّنَ ٱلْعَـٰلَمِينَ ﴿٨٠﴾

    «أَتَأْتُونَ» بدل «إِنَّكُمْ لَتَأْتُونَ»

  9. قَالَ إِنَّ فِيهَا لُوطًا ۚ قَالُوا۟ نَحْنُ أَعْلَمُ بِمَن فِيهَا ۖ لَنُنَجِّيَنَّهُۥ وَأَهْلَهُۥٓ إِلَّا ٱمْرَأَتَهُۥ كَانَتْ مِنَ ٱلْغَـٰبِرِينَ ﴿٣٢﴾

    الأعراف ٨٣

    فَأَنجَيْنَـٰهُ وَأَهْلَهُۥٓ إِلَّا ٱمْرَأَتَهُۥ كَانَتْ مِنَ ٱلْغَـٰبِرِينَ ﴿٨٣﴾

    «فَأَنجَيْنَـٰهُ» بدل «قَالَ إِنَّ فِيهَا … فِيهَا لَنُنَجِّيَنَّهُۥ»

  10. فَكَذَّبُوهُ فَأَخَذَتْهُمُ ٱلرَّجْفَةُ فَأَصْبَحُوا۟ فِى دَارِهِمْ جَـٰثِمِينَ ﴿٣٧﴾

    الأعراف ٧٨، ٩١

    فَأَخَذَتْهُمُ ٱلرَّجْفَةُ فَأَصْبَحُوا۟ فِى دَارِهِمْ جَـٰثِمِينَ ﴿٧٨﴾

    حذف «فَكَذَّبُوهُ» من أولها

  11. خَلَقَ ٱللَّهُ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضَ بِٱلْحَقِّ ۚ إِنَّ فِى ذَٰلِكَ لَـَٔايَةً لِّلْمُؤْمِنِينَ ﴿٤٤﴾

    الحجر ٧٧

    إِنَّ فِى ذَٰلِكَ لَـَٔايَةً لِّلْمُؤْمِنِينَ ﴿٧٧﴾

    حذف «خَلَقَ ٱللَّهُ ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضَ بِٱلْحَقِّ» من أولها

  12. وَقَالُوا۟ لَوْلَآ أُنزِلَ عَلَيْهِ ءَايَـٰتٌ مِّن رَّبِّهِۦ ۖ قُلْ إِنَّمَا ٱلْـَٔايَـٰتُ عِندَ ٱللَّهِ وَإِنَّمَآ أَنَا۠ نَذِيرٌ مُّبِينٌ ﴿٥٠﴾

    الملك ٢٦

    قُلْ إِنَّمَا ٱلْعِلْمُ عِندَ ٱللَّهِ وَإِنَّمَآ أَنَا۠ نَذِيرٌ مُّبِينٌ ﴿٢٦﴾

    تشبه آخرَ آية هذه السورة: «ٱلْعِلْمُ» بدل «ٱلْـَٔايَـٰتُ»

  13. وَيَسْتَعْجِلُونَكَ بِٱلْعَذَابِ ۚ وَلَوْلَآ أَجَلٌ مُّسَمًّى لَّجَآءَهُمُ ٱلْعَذَابُ وَلَيَأْتِيَنَّهُم بَغْتَةً وَهُمْ لَا يَشْعُرُونَ ﴿٥٣﴾

    الشعراء ٢٠٢

    فَيَأْتِيَهُم بَغْتَةً وَهُمْ لَا يَشْعُرُونَ ﴿٢٠٢﴾

    «فَيَأْتِيَهُم» بدل «وَيَسْتَعْجِلُونَكَ بِٱلْعَذَابِ وَلَوْلَآ أَجَلٌ مُّسَمًّى لَّجَآءَهُمُ ٱلْعَذَابُ وَلَيَأْتِيَنَّهُم»

  14. يَسْتَعْجِلُونَكَ بِٱلْعَذَابِ وَإِنَّ جَهَنَّمَ لَمُحِيطَةٌۢ بِٱلْكَـٰفِرِينَ ﴿٥٤﴾

    التوبة ٤٩

    وَمِنْهُم مَّن يَقُولُ ٱئْذَن لِّى وَلَا تَفْتِنِّىٓ ۚ أَلَا فِى ٱلْفِتْنَةِ سَقَطُوا۟ ۗ وَإِنَّ جَهَنَّمَ لَمُحِيطَةٌۢ بِٱلْكَـٰفِرِينَ ﴿٤٩﴾

    «وَمِنْهُم مَّن يَقُولُ … ٱلْفِتْنَةِ سَقَطُوا۟» بدل «يَسْتَعْجِلُونَكَ بِٱلْعَذَابِ»

  15. ٱلَّذِينَ صَبَرُوا۟ وَعَلَىٰ رَبِّهِمْ يَتَوَكَّلُونَ ﴿٥٩﴾

    النحل ٩٩

    إِنَّهُۥ لَيْسَ لَهُۥ سُلْطَـٰنٌ عَلَى ٱلَّذِينَ ءَامَنُوا۟ وَعَلَىٰ رَبِّهِمْ يَتَوَكَّلُونَ ﴿٩٩﴾

    آخرها يشبه آية هذه السورة: «ءَامَنُوا۟» بدل «صَبَرُوا۟»

  16. لِيَكْفُرُوا۟ بِمَآ ءَاتَيْنَـٰهُمْ وَلِيَتَمَتَّعُوا۟ ۖ فَسَوْفَ يَعْلَمُونَ ﴿٦٦﴾

    النحل ٥٥؛ الروم ٣٤

    لِيَكْفُرُوا۟ بِمَآ ءَاتَيْنَـٰهُمْ ۚ فَتَمَتَّعُوا۟ ۖ فَسَوْفَ تَعْلَمُونَ ﴿٥٥﴾

    «فَتَمَتَّعُوا۟» بدل «وَلِيَتَمَتَّعُوا۟»؛ «تَعْلَمُونَ» بدل «يَعْلَمُونَ»

2 · Aleyas similares que difieren en algunas palabras

(1)

  1. 1 2 apariciones; comparadas con la aleya 3

    وَلَقَدْ فَتَنَّا ٱلَّذِينَ مِن قَبْلِهِمْ ۖ فَلَيَعْلَمَنَّ ٱللَّهُ ٱلَّذِينَ صَدَقُوا۟ وَلَيَعْلَمَنَّ ٱلْكَـٰذِبِينَ ﴿٣﴾

    وَلَيَعْلَمَنَّ ٱللَّهُ ٱلَّذِينَ ءَامَنُوا۟ وَلَيَعْلَمَنَّ ٱلْمُنَـٰفِقِينَ ﴿١١﴾

    Se parece al final de la aleya 3: «وَلَيَعْلَمَنَّ» en lugar de «فَلَيَعْلَمَنَّ»; «ءَامَنُوا۟» en lugar de «صَدَقُوا۟»; «ٱلْمُنَـٰفِقِينَ» en lugar de «ٱلْكَـٰذِبِينَ»

Dónde difiere la primera aleya · palabra en lugar de otra · adición · omisión · cambio de orden

3 · Aleyas que comparten una expresión y luego difieren

(3)

  1. 1 2 aleyas comparten «مِن دُونِ ٱللَّهِ أَوْثَـٰنًا»

    إِنَّمَا تَعْبُدُونَ مِن دُونِ ٱللَّهِ أَوْثَـٰنًا وَتَخْلُقُونَ إِفْكًا ۚ إِنَّ ٱلَّذِينَ … ﴿١٧﴾

    وَقَالَ إِنَّمَا ٱتَّخَذْتُم مِّن دُونِ ٱللَّهِ أَوْثَـٰنًا مَّوَدَّةَ بَيْنِكُمْ فِى ٱلْحَيَوٰةِ … ﴿٢٥﴾

    Precedida de «ٱتَّخَذْتُم» en lugar de «تَعْبُدُونَ»; seguida de «مَّوَدَّةَ» en lugar de «وَتَخْلُقُونَ»
