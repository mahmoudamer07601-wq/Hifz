  2. ٢ ٢ مواضع؛ المقارنة بالآية ٥٩

    قَالُوا۟ مَن فَعَلَ هَـٰذَا بِـَٔالِهَتِنَآ إِنَّهُۥ لَمِنَ ٱلظَّـٰلِمِينَ ﴿٥٩﴾

    قَالُوٓا۟ ءَأَنتَ فَعَلْتَ هَـٰذَا بِـَٔالِهَتِنَا يَـٰٓإِبْرَٰهِيمُ ﴿٦٢﴾

  3. ٣ ٢ مواضع؛ المقارنة بالآية ٧٥

٣ · آيات تشترك في عبارة ثم تختلف

(٣)

  1. ١ آيتان تشتركان في «تَعْبُدُونَ مِن دُونِ ٱللَّهِ»

    أُفٍّ لَّكُمْ وَلِمَا تَعْبُدُونَ مِن دُونِ ٱللَّهِ ۖ أَفَلَا تَعْقِلُونَ ﴿٦٧﴾

    إِنَّكُمْ وَمَا تَعْبُدُونَ مِن دُونِ ٱللَّهِ حَصَبُ جَهَنَّمَ أَنتُمْ لَهَا وَٰرِدُونَ ﴿٩٨﴾

    قبلها «وَمَا» بدل «وَلِمَا»؛ بعدها «حَصَبُ» بدل «أَفَلَا»

  2. ٢ آيتان تشتركان في «إِلَى ٱلْأَرْضِ ٱلَّتِى بَـٰرَكْنَا فِيهَا»

    وَنَجَّيْنَـٰهُ وَلُوطًا إِلَى ٱلْأَرْضِ ٱلَّتِى بَـٰرَكْنَا فِيهَا لِلْعَـٰلَمِينَ ﴿٧١﴾

    وَلِسُلَيْمَـٰنَ ٱلرِّيحَ عَاصِفَةً تَجْرِى بِأَمْرِهِۦٓ إِلَى ٱلْأَرْضِ ٱلَّتِى بَـٰرَكْنَا فِيهَا ۚ وَكُنَّا بِكُلِّ شَىْءٍ عَـٰلِمِينَ ﴿٨١﴾

    قبلها «بِأَمْرِهِۦٓ» بدل «وَلُوطًا»؛ بعدها «وَكُنَّا» بدل «لِلْعَـٰلَمِينَ»

  3. ٣ آيتان تشتركان في «إِنَّهُمْ كَانُوا۟ قَوْمَ سَوْءٍ»

    وَلُوطًا ءَاتَيْنَـٰهُ حُكْمًا وَعِلْمًا وَنَجَّيْنَـٰهُ مِنَ ٱلْقَرْيَةِ ٱلَّتِى كَانَت تَّعْمَلُ ٱلْخَبَـٰٓئِثَ ۗ إِنَّهُمْ كَانُوا۟ قَوْمَ سَوْءٍ فَـٰسِقِينَ ﴿٧٤﴾

    وَنَصَرْنَـٰهُ مِنَ ٱلْقَوْمِ ٱلَّذِينَ كَذَّبُوا۟ بِـَٔايَـٰتِنَآ ۚ إِنَّهُمْ كَانُوا۟ قَوْمَ سَوْءٍ فَأَغْرَقْنَـٰهُمْ أَجْمَعِينَ ﴿٧٧﴾

    قبلها «بِـَٔايَـٰتِنَآ» بدل «ٱلْخَبَـٰٓئِثَ»؛ بعدها «فَأَغْرَقْنَـٰهُمْ» بدل «فَـٰسِقِينَ»

    وَمَا خَلَقْنَا ٱلسَّمَـٰوَٰتِ وَٱلْأَرْضَ وَمَا بَيْنَهُمَا لَـٰعِبِينَ ﴿٣٨﴾

    «ٱلسَّمَـٰوَٰتِ» بدل «ٱلسَّمَآءَ»

  4. لَوْ كَانَ فِيهِمَآ ءَالِهَةٌ إِلَّا ٱللَّهُ لَفَسَدَتَا ۚ فَسُبْحَـٰنَ ٱللَّهِ رَبِّ ٱلْعَرْشِ عَمَّا يَصِفُونَ ﴿٢٢﴾

    الصافات ١٨٠

    سُبْحَـٰنَ رَبِّكَ رَبِّ ٱلْعِزَّةِ عَمَّا يَصِفُونَ ﴿١٨٠﴾

    تشبه آخرَ آية هذه السورة: «سُبْحَـٰنَ رَبِّكَ» بدل «فَسُبْحَـٰنَ ٱللَّهِ»؛ «ٱلْعِزَّةِ» بدل «ٱلْعَرْشِ»

  5. وَقَالُوا۟ ٱتَّخَذَ ٱلرَّحْمَـٰنُ وَلَدًا ۗ سُبْحَـٰنَهُۥ ۚ بَلْ عِبَادٌ مُّكْرَمُونَ ﴿٢٦﴾

    مريم ٨٨

    وَقَالُوا۟ ٱتَّخَذَ ٱلرَّحْمَـٰنُ وَلَدًا ﴿٨٨﴾

    حذف «سُبْحَـٰنَهُۥ بَلْ عِبَادٌ مُّكْرَمُونَ» من آخرها

  6. يَعْلَمُ مَا بَيْنَ أَيْدِيهِمْ وَمَا خَلْفَهُمْ وَلَا يَشْفَعُونَ إِلَّا لِمَنِ ٱرْتَضَىٰ وَهُم مِّنْ خَشْيَتِهِۦ مُشْفِقُونَ ﴿٢٨﴾

    طه ١١٠

    يَعْلَمُ مَا بَيْنَ أَيْدِيهِمْ وَمَا خَلْفَهُمْ وَلَا يُحِيطُونَ بِهِۦ عِلْمًا ﴿١١٠﴾

    «يُحِيطُونَ بِهِۦ عِلْمًا» بدل «يَشْفَعُونَ إِلَّا لِمَنِ ٱرْتَضَىٰ وَهُم مِّنْ خَشْيَتِهِۦ مُشْفِقُونَ»

  7. وَيَقُولُونَ مَتَىٰ هَـٰذَا ٱلْوَعْدُ إِن كُنتُمْ صَـٰدِقِينَ ﴿٣٨﴾

    السجدة ٢٨

    وَيَقُولُونَ مَتَىٰ هَـٰذَا ٱلْفَتْحُ إِن كُنتُمْ صَـٰدِقِينَ ﴿٢٨﴾

    «ٱلْفَتْحُ» بدل «ٱلْوَعْدُ»

  8. إِذْ قَالَ لِأَبِيهِ وَقَوْمِهِۦ مَا هَـٰذِهِ ٱلتَّمَاثِيلُ ٱلَّتِىٓ أَنتُمْ لَهَا عَـٰكِفُونَ ﴿٥٢﴾

    الشعراء ٧٠

    إِذْ قَالَ لِأَبِيهِ وَقَوْمِهِۦ مَا تَعْبُدُونَ ﴿٧٠﴾

    «تَعْبُدُونَ» بدل «هَـٰذِهِ ٱلتَّمَاثِيلُ ٱلَّتِىٓ أَنتُمْ لَهَا عَـٰكِفُونَ»

    الصافات ٨٥

    إِذْ قَالَ لِأَبِيهِ وَقَوْمِهِۦ مَاذَا تَعْبُدُونَ ﴿٨٥﴾

    «مَاذَا تَعْبُدُونَ» بدل «مَا هَـٰذِهِ ٱلتَّمَاثِيلُ ٱلَّتِىٓ أَنتُمْ لَهَا عَـٰكِفُونَ»

  9. وَأَرَادُوا۟ بِهِۦ كَيْدًا فَجَعَلْنَـٰهُمُ ٱلْأَخْسَرِينَ ﴿٧٠﴾

    الصافات ٩٨

    فَأَرَادُوا۟ بِهِۦ كَيْدًا فَجَعَلْنَـٰهُمُ ٱلْأَسْفَلِينَ ﴿٩٨﴾

    «فَأَرَادُوا۟» بدل «وَأَرَادُوا۟»؛ «ٱلْأَسْفَلِينَ» بدل «ٱلْأَخْسَرِينَ»

  10. وَنُوحًا إِذْ نَادَىٰ مِن قَبْلُ فَٱسْتَجَبْنَا لَهُۥ فَنَجَّيْنَـٰهُ وَأَهْلَهُۥ مِنَ ٱلْكَرْبِ ٱلْعَظِيمِ ﴿٧٦﴾

    الصافات ٧٦

    وَنَجَّيْنَـٰهُ وَأَهْلَهُۥ مِنَ ٱلْكَرْبِ ٱلْعَظِيمِ ﴿٧٦﴾

    تشبه آخرَ آية هذه السورة: «وَنَجَّيْنَـٰهُ» بدل «فَنَجَّيْنَـٰهُ»

  11. إِنَّ هَـٰذِهِۦٓ أُمَّتُكُمْ أُمَّةً وَٰحِدَةً وَأَنَا۠ رَبُّكُمْ فَٱعْبُدُونِ ﴿٩٢﴾

    المؤمنون ٥٢

    وَإِنَّ هَـٰذِهِۦٓ أُمَّتُكُمْ أُمَّةً وَٰحِدَةً وَأَنَا۠ رَبُّكُمْ فَٱتَّقُونِ ﴿٥٢﴾

    «وَإِنَّ» بدل «إِنَّ»؛ «فَٱتَّقُونِ» بدل «فَٱعْبُدُونِ»

  12. وَتَقَطَّعُوٓا۟ أَمْرَهُم بَيْنَهُمْ ۖ كُلٌّ إِلَيْنَا رَٰجِعُونَ ﴿٩٣﴾

    المؤمنون ٥٣

    فَتَقَطَّعُوٓا۟ أَمْرَهُم بَيْنَهُمْ زُبُرًا ۖ كُلُّ حِزْبٍۭ بِمَا لَدَيْهِمْ فَرِحُونَ ﴿٥٣﴾

    «فَتَقَطَّعُوٓا۟» بدل «وَتَقَطَّعُوٓا۟»؛ «زُبُرًا كُلُّ حِزْبٍۭ بِمَا لَدَيْهِمْ فَرِحُونَ» بدل «كُلٌّ إِلَيْنَا رَٰجِعُونَ»

  13. وَمَآ أَرْسَلْنَـٰكَ إِلَّا رَحْمَةً لِّلْعَـٰلَمِينَ ﴿١٠٧﴾

    الفرقان ٥٦

    وَمَآ أَرْسَلْنَـٰكَ إِلَّا مُبَشِّرًا وَنَذِيرًا ﴿٥٦﴾

    «مُبَشِّرًا وَنَذِيرًا» بدل «رَحْمَةً لِّلْعَـٰلَمِينَ»

3 · Aleyas que comparten una expresión y luego difieren

(3)

  1. 1 2 aleyas comparten «تَعْبُدُونَ مِن دُونِ ٱللَّهِ»

    أُفٍّ لَّكُمْ وَلِمَا تَعْبُدُونَ مِن دُونِ ٱللَّهِ ۖ أَفَلَا تَعْقِلُونَ ﴿٦٧﴾

    إِنَّكُمْ وَمَا تَعْبُدُونَ مِن دُونِ ٱللَّهِ حَصَبُ جَهَنَّمَ أَنتُمْ لَهَا وَٰرِدُونَ ﴿٩٨﴾

    Precedida de «وَمَا» en lugar de «وَلِمَا»; seguida de «حَصَبُ» en lugar de «أَفَلَا»

  2. 2 2 aleyas comparten «إِلَى ٱلْأَرْضِ ٱلَّتِى بَـٰرَكْنَا فِيهَا»

    وَنَجَّيْنَـٰهُ وَلُوطًا إِلَى ٱلْأَرْضِ ٱلَّتِى بَـٰرَكْنَا فِيهَا لِلْعَـٰلَمِينَ ﴿٧١﴾

    وَلِسُلَيْمَـٰنَ ٱلرِّيحَ عَاصِفَةً تَجْرِى بِأَمْرِهِۦٓ إِلَى ٱلْأَرْضِ ٱلَّتِى بَـٰرَكْنَا فِيهَا ۚ وَكُنَّا بِكُلِّ شَىْءٍ عَـٰلِمِينَ ﴿٨١﴾

    Precedida de «بِأَمْرِهِۦٓ» en lugar de «وَلُوطًا»; seguida de «وَكُنَّا» en lugar de «لِلْعَـٰلَمِينَ»

  3. 3 2 aleyas comparten «إِنَّهُمْ كَانُوا۟ قَوْمَ سَوْءٍ»

    وَلُوطًا ءَاتَيْنَـٰهُ حُكْمًا وَعِلْمًا وَنَجَّيْنَـٰهُ مِنَ ٱلْقَرْيَةِ ٱلَّتِى كَانَت تَّعْمَلُ ٱلْخَبَـٰٓئِثَ ۗ إِنَّهُمْ كَانُوا۟ قَوْمَ سَوْءٍ فَـٰسِقِينَ ﴿٧٤﴾

    وَنَصَرْنَـٰهُ مِنَ ٱلْقَوْمِ ٱلَّذِينَ كَذَّبُوا۟ بِـَٔايَـٰتِنَآ ۚ إِنَّهُمْ كَانُوا۟ قَوْمَ سَوْءٍ فَأَغْرَقْنَـٰهُمْ أَجْمَعِينَ ﴿٧٧﴾

    Precedida de «بِـَٔايَـٰتِنَآ» en lugar de «ٱلْخَبَـٰٓئِثَ»; seguida de «فَأَغْرَقْنَـٰهُمْ» en lugar de «فَـٰسِقِينَ»

فَمَن يَعْمَلْ مِنَ ٱلصَّـٰلِحَـٰتِ وَهُوَ مُؤْمِنٌ فَلَا كُفْرَانَ لِسَعْيِهِۦ وَإِنَّا لَهُۥ كَـٰتِبُونَ ﴿٩٤﴾ وَحَرَٰمٌ عَلَىٰ قَرْيَةٍ أَهْلَكْنَـٰهَآ أَنَّهُمْ لَا يَرْجِعُونَ ﴿٩٥﴾ حَتَّىٰٓ إِذَا فُتِحَتْ يَأْجُوجُ وَمَأْجُوجُ وَهُم مِّن كُلِّ حَدَبٍ يَنسِلُونَ ﴿٩٦﴾ وَٱقْتَرَبَ ٱلْوَعْدُ ٱلْحَقُّ فَإِذَا هِىَ شَـٰخِصَةٌ أَبْصَـٰرُ ٱلَّذِينَ كَفَرُوا۟ يَـٰوَيْلَنَا قَدْ كُنَّا فِى غَفْلَةٍ مِّنْ هَـٰذَا بَلْ كُنَّا ظَـٰلِمِينَ ﴿٩٧﴾ إِنَّكُمْ وَمَا تَعْبُدُونَ مِن دُونِ ٱللَّهِ حَصَبُ جَهَنَّمَ أَنتُمْ لَهَا وَٰرِدُونَ ﴿٩٨﴾ لَوْ كَانَ هَـٰٓؤُلَآءِ ءَالِهَةً مَّا وَرَدُوهَا ۖ وَكُلٌّ فِيهَا خَـٰلِدُونَ ﴿٩٩﴾ لَهُمْ فِيهَا زَفِيرٌ وَهُمْ فِيهَا لَا يَسْمَعُونَ ﴿١٠٠﴾ إِنَّ ٱلَّذِينَ سَبَقَتْ لَهُم مِّنَّا ٱلْحُسْنَىٰٓ أُو۟لَـٰٓئِكَ عَنْهَا مُبْعَدُونَ ﴿١٠١﴾ لَا يَسْمَعُونَ حَسِيسَهَا ۖ وَهُمْ فِى مَا ٱشْتَهَتْ أَنفُسُهُمْ خَـٰلِدُونَ ﴿١٠٢﴾ لَا يَحْزُنُهُمُ ٱلْفَزَعُ ٱلْأَكْبَرُ وَتَتَلَقَّىٰهُمُ ٱلْمَلَـٰٓئِكَةُ هَـٰذَا يَوْمُكُمُ ٱلَّذِى كُنتُمْ تُوعَدُونَ ﴿١٠٣﴾ يَوْمَ نَطْوِى ٱلسَّمَآءَ كَطَىِّ ٱلسِّجِلِّ لِلْكُتُبِ ۚ كَمَا بَدَأْنَآ أَوَّلَ خَلْقٍ نُّعِيدُهُۥ ۚ وَعْدًا عَلَيْنَآ ۚ إِنَّا كُنَّا فَـٰعِلِينَ ﴿١٠٤﴾

«ولقد كتبنا في الزبور من بعد الذكر»: «أنّ الأرض يرثها عبادي الصالحون»، «وما أرسلناك إلّا رحمة للعالمين»، «فإن تولّوا فقل آذنتكم على سواء»، و«قال ربّ احكم بالحقّ»

وَلَقَدْ كَتَبْنَا فِى ٱلزَّبُورِ مِنۢ بَعْدِ ٱلذِّكْرِ أَنَّ ٱلْأَرْضَ يَرِثُهَا عِبَادِىَ ٱلصَّـٰلِحُونَ ﴿١٠٥﴾ إِنَّ فِى هَـٰذَا لَبَلَـٰغًا لِّقَوْمٍ عَـٰبِدِينَ ﴿١٠٦﴾ وَمَآ أَرْسَلْنَـٰكَ إِلَّا رَحْمَةً لِّلْعَـٰلَمِينَ ﴿١٠٧﴾ قُلْ إِنَّمَا يُوحَىٰٓ إِلَىَّ أَنَّمَآ إِلَـٰهُكُمْ إِلَـٰهٌ وَٰحِدٌ ۖ فَهَلْ أَنتُم مُّسْلِمُونَ ﴿١٠٨﴾ فَإِن تَوَلَّوْا۟ فَقُلْ ءَاذَنتُكُمْ عَلَىٰ سَوَآءٍ ۖ وَإِنْ أَدْرِىٓ أَقَرِيبٌ أَم بَعِيدٌ مَّا تُوعَدُونَ ﴿١٠٩﴾ إِنَّهُۥ يَعْلَمُ ٱلْجَهْرَ مِنَ ٱلْقَوْلِ وَيَعْلَمُ مَا تَكْتُمُونَ ﴿١١٠﴾ وَإِنْ أَدْرِى لَعَلَّهُۥ فِتْنَةٌ لَّكُمْ وَمَتَـٰعٌ إِلَىٰ حِينٍ ﴿١١١﴾ قَـٰلَ رَبِّ ٱحْكُم بِٱلْحَقِّ ۗ وَرَبُّنَا ٱلرَّحْمَـٰنُ ٱلْمُسْتَعَانُ عَلَىٰ مَا تَصِفُونَ ﴿١١٢﴾
